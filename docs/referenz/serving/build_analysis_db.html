serving.build_analysis_db

Baut data/serving/analyse.duckdb - die Auswertungsumgebung.

Aufruf aus dem Projekt-Root::

python -m serving.build_analysis_db

Danach::

duckdb data/serving/analyse.duckdb
D SELECT city, nibrs_description, count(*) FROM crime_offense
  GROUP BY 1, 2 ORDER BY 3 DESC LIMIT 10;

Was das Skript tut: Es sucht jede Tabelle unter data/silver/ und data/gold/, legt je eine Sicht darauf an und schreibt dazu zwei Hilfssichten - herkunft und tabellen. Die Datei haelt keine Daten; warum das so ist, steht im Modulkopf von serving.

Drei Eigenheiten, die beim Bauen zu beachten waren:

  1. Absolute Pfade. DuckDB loest relative Pfade gegen das Arbeitsverzeichnis des Prozesses auf, nicht gegen den Ort der Datenbank. Eine Sicht mit relativem Pfad funktionierte also nur, solange man die Datei aus dem Projekt-Root oeffnet. Die Datei ist ohnehin abgeleitet und nicht versioniert, deshalb sind absolute Pfade der ehrlichere Weg: Wer das Projekt verschiebt, baut sie neu.
  2. Vintages. tract liegt unter dataset=tract/vintage=2024/. Die Sicht zeigt auf die juengste Vintage und nennt sie in tabellen, damit niemand raten muss, welchen Stand er sieht.
  3. Atomar veroeffentlicht. Geschrieben wird in eine .part-Datei, die erst nach dem letzten Schritt umbenannt wird - sonst haette ein abgebrochener Lauf eine halbe Datenbank hinterlassen, in der die Haelfte der Sichten fehlt.

 56# Die Schichten, die abfragbar gemacht werden. Beide sind gebaut: vier
 57# Silver-Tabellen und die fuenf des Sternschemas, zusammen neun Sichten. Als

 93def discover(data_root: Path) -> list[Tabelle]:
 94    """Findet jede Tabelle unter den Schichten.
 95
 96    Gesucht wird nach ``dataset=<name>/**/part-0001.parquet``. Liegen mehrere
 97    Vintages vor, gewinnt die letzte in Sortierreihenfolge - bei ``vintage=``
 98    ist das die juengste.
 99    """
100    gefunden: dict[str, Tabelle] = {}
101    ablage = for_root(data_root)
102    for layer in SCHICHTEN:
103        for rel in ablage.list_files(layer, f"dataset=*/**/{PARQUET_FILE}"):
104            name = None
105            partition = ""
106            for teil in rel.split("/")[1:]:
107                if teil.startswith("dataset="):
108                    name = teil.removeprefix("dataset=")
109                elif "=" in teil:
110                    partition = teil
111            if name is None:
112                continue
113            # Lokal absolut, damit die Datenbank von ueberall lesbar bleibt;
114            # auf S3 die URL. Im Hybridmodus entscheidet die Ablage je Datei.
115            pfad = ablage.resolve_read_path(rel)
116            path: str | Path = pfad if pfad.startswith("s3://") else Path(pfad).resolve()
117            # Spaetere Schichten und juengere Vintages ueberschreiben.
118            gefunden[name] = Tabelle(name, layer, path, partition)
119    return sorted(gefunden.values(), key=lambda t: (t.layer, t.name))

122def create_views(con: duckdb.DuckDBPyConnection, tabellen: list[Tabelle]) -> None:
123    """Legt je Tabelle eine Sicht an, plus zwei Hilfssichten.
124
125    ``herkunft`` und ``tabellen`` sind das, was eine Auswertungsumgebung von
126    einem Haufen Dateien unterscheidet: Man kann sie fragen, was man vor sich
127    hat, ohne das Repository zu kennen.
128    """
129    for t in tabellen:
130        try:
131            con.execute(
132                f'CREATE OR REPLACE VIEW "{t.name}" AS '
133                f"SELECT * FROM read_parquet('{t.path}')"
134            )
135        except duckdb.Error as fehler:
136            # DuckDB prueft die Datei schon beim Anlegen der Sicht. Der Fehler
137            # bekommt hier denselben Wortlaut wie in verify(), damit der
138            # Aufrufer nicht zwei Formulierungen fuer dieselbe Sache kennen muss.
139            raise RuntimeError(
140                f"Sicht {t.name} ist nicht lesbar: {t.path}: {fehler}"
141            ) from fehler
142        LOG.info("Sicht %s -> %s", t.name, t.path)
143
144    werte = ", ".join(
145        f"('{t.name}', '{t.layer}', '{t.partition}', '{t.path}')" for t in tabellen
146    )
147    con.execute(f"""
148        CREATE OR REPLACE VIEW tabellen AS
149        SELECT * FROM (VALUES {werte}) AS t(name, schicht, partition, datei)
150    """)
151
152    gebaut = datetime.now(timezone.utc).isoformat()
153    con.execute(f"""
154        CREATE OR REPLACE VIEW herkunft AS
155        SELECT * FROM (VALUES
156            ('gebaut_am_utc', '{gebaut}'),
157            ('duckdb', '{duckdb.__version__}'),
158            ('tabellen', '{len(tabellen)}'),
159            ('hinweis', 'Sichten ueber Parquet. Die Datei haelt keine Daten; '
160                        'ein Silver-Neubau ist sofort sichtbar.'),
161            ('eigene_tabellen', 'CREATE TABLE meins AS SELECT ... ist erlaubt '
162                                'und liegt neben den Sichten.')
163        ) AS h(schluessel, wert)
164    """)

167def verify(con: duckdb.DuckDBPyConnection, tabellen: list[Tabelle]) -> dict[str, int]:
168    """Jede Sicht muss lesbar sein und Zeilen liefern.
169
170    Eine Sicht auf eine Datei, die es nicht mehr gibt, faellt erst beim ersten
171    ``SELECT`` auf - also bei dem, der auswerten will. Lieber hier.
172    """
173    zeilen: dict[str, int] = {}
174    for t in tabellen:
175        try:
176            zeilen[t.name] = con.execute(
177                f'SELECT count(*) FROM "{t.name}"'
178            ).fetchone()[0]
179        except duckdb.Error as fehler:
180            raise RuntimeError(
181                f"Sicht {t.name} ist nicht lesbar: {fehler}"
182            ) from fehler
183    return zeilen

186def uebernimm_eigene_tabellen(con: duckdb.DuckDBPyConnection, alt: Path) -> list[str]:
187    """Kopiert selbst angelegte Tabellen aus der bisherigen Datenbank herueber.
188
189    **Warum das noetig ist.** Konzept 5.2 laedt ausdruecklich dazu ein: *"wer
190    ein Zwischenergebnis behalten will, legt daneben ``CREATE TABLE meins AS
191    SELECT …`` an"*, und die Hinweistabelle der Datenbank sagt dasselbe. Nur
192    entstand beim Neubau eine **neue Datei**, die die bisherige ersetzte - die
193    eigene Tabelle war weg, ohne Meldung. Eine externe Pruefung hat das am
194    15.9.2026 vorgefuehrt; die Einladung stand da seit dem ersten Tag.
195
196    Uebernommen wird, was eine echte Tabelle ist. Alles, was dieses Modul
197    anlegt, sind Sichten - was also als Tabelle dasteht, hat jemand selbst
198    angelegt. Traegt eine davon den Namen einer Sicht, gewinnt die Sicht und
199    die Tabelle wird mit Meldung uebergangen: Sonst verdeckte ein
200    Zwischenergebnis stillschweigend die Daten, ueber die es gerechnet wurde.
201    """
202    if not alt.is_file():
203        return []
204    eigene: list[str] = []
205    con.execute(f"ATTACH '{alt}' AS vorher (READ_ONLY)")
206    try:
207        sichten = {r[0] for r in con.execute(
208            "SELECT view_name FROM duckdb_views() WHERE NOT internal"
209        ).fetchall()}
210        namen = [r[0] for r in con.execute(
211            "SELECT table_name FROM duckdb_tables() "
212            "WHERE database_name = 'vorher' AND NOT internal ORDER BY table_name"
213        ).fetchall()]
214        for name in namen:
215            if name in sichten:
216                LOG.warning(
217                    "Eigene Tabelle %r nicht uebernommen: Eine Sicht dieses "
218                    "Namens gehoert zur Auswertung. Erst umbenennen, dann neu "
219                    "bauen.", name,
220                )
221                continue
222            con.execute(f'CREATE TABLE "{name}" AS SELECT * FROM vorher."{name}"')
223            eigene.append(name)
224    except duckdb.Error as fehler:
225        LOG.warning("Eigene Tabellen nicht uebernommen: %s", fehler)
226    finally:
227        con.execute("DETACH vorher")
228    return eigene

231def build(data_root: Path) -> Path:
232    tabellen = discover(data_root)
233    if not tabellen:
234        raise RuntimeError(
235            f"Keine Tabellen unter {data_root}/silver oder {data_root}/gold. "
236            "Zuerst die Silver-Schicht bauen, etwa "
237            "'python -m silver_pipeline.build_crime_offense'."
238        )
239
240    ziel = database_path(data_root)
241    ziel.parent.mkdir(parents=True, exist_ok=True)
242    arbeit = ziel.with_name(f".{ziel.name}.part")
243    arbeit.unlink(missing_ok=True)
244
245    # Die Datenbank selbst bleibt immer lokal (data/serving/), in jedem Modus.
246    # Kann die Ablage S3 lesen, bringt die Verbindung httpfs und ein
247    # temporaeres Secret mit - und wer die Datei spaeter oeffnet, braucht
248    # dasselbe: storage.duckdb.connect(config, pfad).
249    con = storage_duckdb.connect(for_root(data_root).config, arbeit)
250    try:
251        # Fuer die Geometrie in tract. Fehlt die Extension, bleiben die Sichten
252        # trotzdem lesbar - nur ST_-Funktionen stehen dann nicht bereit.
253        try:
254            con.execute("INSTALL spatial")
255            con.execute("LOAD spatial")
256        except duckdb.Error as fehler:
257            LOG.warning("spatial nicht verfuegbar, Geometrie bleibt WKB: %s", fehler)
258
259        create_views(con, tabellen)
260        zeilen = verify(con, tabellen)
261        uebernommen = uebernimm_eigene_tabellen(con, ziel)
262    except BaseException:
263        # Eine halbe Datenbank ist schlimmer als keine: In ihr fehlt ein Teil
264        # der Sichten, ohne dass man es ihr ansieht.
265        con.close()
266        arbeit.unlink(missing_ok=True)
267        raise
268    finally:
269        con.close()
270
271    arbeit.replace(ziel)
272    for name, n in zeilen.items():
273        LOG.info("  %-18s %8d Zeilen", name, n)
274    if uebernommen:
275        LOG.info("  eigene Tabellen uebernommen: %s", ", ".join(uebernommen))
276    LOG.info("%s: %d Sichten, %.1f KB",
277             ziel, len(tabellen), ziel.stat().st_size / 1024)
278    return ziel

281def build_selbststaendig(
282    tabellen: list[tuple[str, Path]],
283    ziel: Path,
284    release: dict[str, Any],
285) -> Path:
286    """Eine Datenbank mit **Tabellen** statt Sichten - fuer ein Release.
287
288    Alles oben gilt weiter: Die Arbeitsdatenbank haelt Sichten, weil Sichten
289    nie veralten und keinen zweiten Bestand bilden. Ein Release ist etwas
290    anderes. Es ist unveraenderlich - eine Kopie darin kann nicht veralten,
291    weil sich nichts darunter mehr bewegt - und es wird auf einen anderen
292    Rechner geholt, wo Sichten mit absoluten Pfaden ins Leere zeigten.
293
294    ``release_info`` traegt Kennung und Zeitpunkt, damit ein Dashboard sagen
295    kann, welchen Stand es zeigt (Rat vom 17.9.2026, Schritt 6).
296    """
297    ziel.parent.mkdir(parents=True, exist_ok=True)
298    arbeit = ziel.with_name(f".{ziel.name}.part")
299    arbeit.unlink(missing_ok=True)
300    con = storage_duckdb.verbinde(arbeit)
301    try:
302        try:
303            con.execute("INSTALL spatial")
304            con.execute("LOAD spatial")
305        except duckdb.Error as fehler:
306            LOG.warning("spatial nicht verfuegbar, Geometrie bleibt WKB: %s", fehler)
307        for name, parquet in tabellen:
308            con.execute(
309                f'CREATE TABLE "{name}" AS SELECT * FROM read_parquet(?)',
310                [str(parquet)],
311            )
312        con.execute(
313            "CREATE TABLE release_info AS SELECT ? AS release_id, ? AS created_at_utc",
314            [release["release_id"], release["created_at_utc"]],
315        )
316    except BaseException:
317        con.close()
318        arbeit.unlink(missing_ok=True)
319        raise
320    finally:
321        con.close()
322    arbeit.replace(ziel)
323    LOG.info("%s: %d Tabellen, %.1f MB", ziel, len(tabellen), ziel.stat().st_size / 1e6)
324    return ziel

327def main() -> int:
328    args = arguments()

331        format="%(levelname)s %(message)s",
332    )

339    return 0

308        for name, parquet in tabellen:
