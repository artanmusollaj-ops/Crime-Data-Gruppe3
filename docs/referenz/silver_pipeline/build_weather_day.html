silver_pipeline.build_weather_day

Baut die Silver-Tabelle weather_day aus den NOAA-Stationsreihen.

Eine Zeile je Stadt und Tag. Das ist eine andere Granularität als die Faktentabelle, und genau deshalb ist es eine eigene Tabelle: Ein Wetterwert an jedes Delikt derselben Stadt und desselben Tages zu haengen wuerde Zeilen vervielfachen, und jede spaetere Aggregation ueber Wetter zaehlte falsch. Verbunden wird erst in Gold, dort auf dem Aggregationsebene.

Aufruf aus dem Projekt-Root:

python -m silver_pipeline.build_weather_day

Was hier gedeutet wird und in Bronze bewusst nicht:

  1. Einheiten. GHCN-Daily fuehrt Temperatur und Niederschlag in Zehnteln, Schneefall dagegen direkt in Millimetern. " 214" sind 21,4 Grad Celsius. An echten Rekorden geprueft: Das Maximum 400 ergibt 40,0 Grad, das Minimum -328 ergibt -32,8 Grad, der Tagesniederschlag 1742 ergibt 174,2 Millimeter, der Schneefall 472 bleibt 472 Millimeter.
  2. Fehlwerte. Ein leeres Feld ist keine Null. Bei Niederschlag ist das der entscheidende Unterschied: kein Messwert ist nicht dasselbe wie kein Regen.
  3. Spurenmengen. Das Messflag T heisst "messbar vorhanden, aber unter der Auflösung". Der Wert steht dann als 0 da, was fachlich etwas anderes ist als trocken. Deshalb eine eigene Spalte.
  4. Qualitaetsflags. Die zweite Position von M,Q,S ist NOAAs Beanstandung. Sie ist fast immer leer, aber nicht immer: O'Hare hat vier SNOW-Zeilen mit I zwischen 1999 und 2011. Beanstandete Werte werden benannt und nicht entfernt.
  5. Eine Station je Stadt. Absprache 3 hat das so entschieden. Diese Tabelle nimmt deshalb je Stadt die Flughafenstation und nennt im Manifest, welche das war und welche geladenen Stationen uebergangen wurden - damit ein zugeschaltetes Midway nicht stillschweigend verschwindet.

Die Feldzahl der _ATTRIBUTES-Spalten wird nicht unterstellt: Sie ist je Groesse und je Station verschieden - drei bei TMAX, vier bei PRCP, und bei Detroits SNOW wieder drei. split_part liefert bei fehlendem Feld einen leeren Text, was hier zu NULL wird; ein Parser mit fester Feldzahl wuerde stattdessen Positionen verschieben.

129def newest_bronze_run(data_root: Path, station: str) -> Source:
130    """Der juengste vollstaendige Bronze-Lauf dieser Station."""
131    ablage = for_root(data_root)
132    prefix = f"{BRONZE_LAYER}/dataset={NOAA_DATASET}/station={station}"
133    lauf = newest_complete_run(ablage, prefix, what=f"{NOAA_DATASET}, Station {station}")
134    stadt = STATIONS.get(station, {}).get("city")
135    if stadt is None:
136        raise TransformError(f"Unbekannte Station: {station}")
137    return Source(station, stadt, lauf.verzeichnis.name,
138                  ablage.resolve_read_path(lauf.relativ, PARQUET_FILE))

141def collect_sources(data_root: Path, stations: list[str]) -> list[Source]:
142    """Je Stadt genau eine Station - sonst bricht die Granularität.
143
144    Absprache 3 hat "eine Station je Stadt" entschieden. Zwei Stationen
145    derselben Stadt wuerden zwei Zeilen fuer denselben Tag ergeben; das faellt
146    hier auf und nicht erst beim Zaehlen in Gold.
147    """
148    quellen = [newest_bronze_run(data_root, station) for station in stations]
149    je_stadt: dict[str, list[Source]] = {}
150    for quelle in quellen:
151        je_stadt.setdefault(quelle.city, []).append(quelle)
152    doppelt = {
153        stadt: [q.station for q in liste]
154        for stadt, liste in je_stadt.items()
155        if len(liste) > 1
156    }
157    if doppelt:
158        raise TransformError(
159            "Mehr als eine Station je Stadt, die Granularität ist Stadt und Tag: "
160            f"{doppelt}. Absprache 3 sieht eine Station je Stadt vor."
161        )
162    return quellen

165def skipped_stations(data_root: Path, used: list[Source]) -> list[dict[str, str]]:
166    """Geladene Stationen, die diese Tabelle uebergeht.
167
168    Sie gehoeren ins Manifest: Wer Midway zuschaltet und die Stadtzeile
169    unveraendert sieht, soll nachlesen koennen, warum.
170    """
171    genutzt = {quelle.station for quelle in used}
172    ablage = for_root(data_root)
173    prefix = f"{BRONZE_LAYER}/dataset={NOAA_DATASET}"
174    stationen = sorted({
175        lauf.relativ[len(prefix) + 1:].split("/")[0].removeprefix("station=")
176        for lauf in ablage.laeufe(prefix, "station=*/run_*/manifest.json")
177    })
178    uebergangen = []
179    for station in stationen:
180        if station in genutzt:
181            continue
182        uebergangen.append(
183            {
184                "station_id": station,
185                "city": STATIONS.get(station, {}).get("city", "unbekannt"),
186                "reason": "je Stadt wird eine Station verwendet (Absprache 3)",
187            }
188        )
189    return uebergangen

192def measurement_sql(column: str, *, tenths: bool) -> str:
193    """Wert und Flags einer Messgroesse in Silver-Form.
194
195    Der Teiler steht hier und nicht in der Auswertung, damit die Einheit an
196    genau einer Stelle festgelegt ist.
197    """
198    teiler = f" / {TENTHS}" if tenths else ""
199    return f"TRY_CAST({column} AS DOUBLE){teiler}"

202def flag_sql(column: str, position: int) -> str:
203    """Eine Position aus ``M,Q,S[,Beobachtungszeit]``.
204
205    ``split_part`` liefert bei fehlendem Feld einen leeren Text - deshalb
206    ``nullif``. Die Feldzahl wird nicht unterstellt, sie ist je Groesse und je
207    Station verschieden.
208    """
209    return f"nullif(split_part({column}_ATTRIBUTES, ',', {position}), '')"

212def build_query(sources: list[Source]) -> str:
213    """Die Abfrage, die aus Stationsreihen Stadt-Tage macht."""
214    lesen = " UNION ALL ".join(
215        f"SELECT '{quelle.city}' AS city, '{quelle.station}' AS station_id, * "
216        f"FROM read_parquet('{quelle.parquet}')"
217        for quelle in sources
218    )
219    return f"""
220    WITH roh AS ({lesen})
221    SELECT
222        city,
223        CAST(DATE AS DATE)                          AS date,
224        station_id,
225        NAME                                        AS station_name,
226        {measurement_sql('TMAX', tenths=True)}      AS temp_max_c,
227        {measurement_sql('TMIN', tenths=True)}      AS temp_min_c,
228        {measurement_sql('PRCP', tenths=True)}      AS precipitation_mm,
229        {measurement_sql('SNOW', tenths=False)}     AS snowfall_mm,
230        -- Spurenmenge: messbar vorhanden, aber unter der Auflösung. Der Wert
231        -- steht als 0 da, was fachlich etwas anderes ist als trocken.
232        coalesce({flag_sql('PRCP', 1)} = '{TRACE_FLAG}', FALSE) AS precipitation_trace,
233        coalesce({flag_sql('SNOW', 1)} = '{TRACE_FLAG}', FALSE) AS snowfall_trace,
234        list_filter([
235            CASE WHEN TRY_CAST(TMAX AS DOUBLE) IS NULL
236                   OR TRY_CAST(TMIN AS DOUBLE) IS NULL
237                 THEN 'MISSING_TEMPERATURE' END,
238            CASE WHEN TRY_CAST(PRCP AS DOUBLE) IS NULL
239                 THEN 'MISSING_PRECIPITATION' END,
240            CASE WHEN TRY_CAST(SNOW AS DOUBLE) IS NULL
241                 THEN 'MISSING_SNOWFALL' END,
242            -- Zweite Position von M,Q,S ist NOAAs Beanstandung. Fast immer
243            -- leer, aber nicht immer.
244            CASE WHEN {flag_sql('TMAX', 2)} IS NOT NULL THEN 'SUSPECT_TEMP_MAX' END,
245            CASE WHEN {flag_sql('TMIN', 2)} IS NOT NULL THEN 'SUSPECT_TEMP_MIN' END,
246            CASE WHEN {flag_sql('PRCP', 2)} IS NOT NULL THEN 'SUSPECT_PRECIPITATION' END,
247            CASE WHEN {flag_sql('SNOW', 2)} IS NOT NULL THEN 'SUSPECT_SNOWFALL' END
248        ], x -> x IS NOT NULL)                      AS quality_flags
249    FROM roh
250    ORDER BY city, date
251    """

254def reconcile(
255    con: duckdb.DuckDBPyConnection, relation: str, sources: list[Source]
256) -> dict[str, Any]:
257    """Weist nach, dass die Granularität haelt und keine Zeile verloren ging."""
258    quellzeilen = sum(
259        con.execute(
260            f"SELECT count(*) FROM read_parquet('{quelle.parquet}')"
261        ).fetchone()[0]
262        for quelle in sources
263    )
264    zeilen = con.execute(f"SELECT count(*) FROM {relation}").fetchone()[0]
265    doppelte = con.execute(
266        f"SELECT count(*) FROM (SELECT city, date FROM {relation} "
267        "GROUP BY city, date HAVING count(*) > 1)"
268    ).fetchone()[0]
269    ohne_datum = con.execute(
270        f"SELECT count(*) FROM {relation} WHERE date IS NULL"
271    ).fetchone()[0]
272
273    if doppelte:
274        raise TransformError(
275            f"{doppelte} Kombinationen aus Stadt und Tag kommen mehrfach vor; "
276            "die Granularität dieser Tabelle ist Stadt und Tag"
277        )
278    if ohne_datum:
279        raise TransformError(f"{ohne_datum} Zeilen ohne lesbares Datum")
280    if zeilen != quellzeilen:
281        raise TransformError(
282            f"Zeilenzahl weicht ab: {quellzeilen} in Bronze, {zeilen} in Silver"
283        )
284
285    return {
286        "verified": True,
287        "bronze_rows": quellzeilen,
288        "row_count": zeilen,
289        "city_date_unique": True,
290        "rows_without_date": ohne_datum,
291    }

294def coverage(con: duckdb.DuckDBPyConnection, relation: str) -> list[dict[str, Any]]:
295    """Abdeckung je Stadt, samt der Luecken in der taeglichen Reihe.
296
297    Die Reihe soll taeglich sein, ist es aber nicht ueberall: Fehlende Tage
298    werden gezaehlt und ausgewiesen, statt sie in einer Auswertung als
299    Nullwerte zu erscheinen.
300    """
301    return [
302        {
303            "city": row[0],
304            "station_id": row[1],
305            "first_date": row[2].isoformat(),
306            "last_date": row[3].isoformat(),
307            "row_count": row[4],
308            "expected_days": row[5],
309            "missing_days": row[5] - row[4],
310            "temperature_filled": row[6],
311            "precipitation_filled": row[7],
312            "trace_precipitation_days": row[8],
313            "suspect_rows": row[9],
314            # Gemessen wird die Dichte zwischen erstem und letztem Tag im
315            # Fenster, nicht der Abstand zum Stichtag: Sonst zaehlte eine
316            # Reihe, die erst 2024 beginnt, zwei Jahrzehnte Abwesenheit als
317            # Luecken. Dass sie spaet beginnt, verraet first_date.
318            "since_cutoff": {
319                "cutoff": COVERAGE_CUTOFF,
320                "row_count": row[10],
321                "first_date": row[11].isoformat() if row[11] else None,
322                "last_date": row[12].isoformat() if row[12] else None,
323                "expected_days": row[13],
324                "missing_days": (row[13] - row[10]) if row[13] else 0,
325                "temperature_filled": row[14],
326            },
327        }
328        for row in con.execute(
329            f"""
330            SELECT city, any_value(station_id), min(date), max(date), count(*),
331                   datediff('day', min(date), max(date)) + 1,
332                   count(temp_max_c),
333                   count(precipitation_mm),
334                   sum(CASE WHEN precipitation_trace THEN 1 ELSE 0 END),
335                   sum(CASE WHEN len(list_filter(quality_flags,
336                        x -> starts_with(x, 'SUSPECT_'))) > 0 THEN 1 ELSE 0 END),
337                   -- dieselben Kennzahlen nochmal ab dem Stichtag
338                   count(*) FILTER (date >= DATE '{COVERAGE_CUTOFF}'),
339                   min(date) FILTER (date >= DATE '{COVERAGE_CUTOFF}'),
340                   max(date) FILTER (date >= DATE '{COVERAGE_CUTOFF}'),
341                   datediff('day',
342                       min(date) FILTER (date >= DATE '{COVERAGE_CUTOFF}'),
343                       max(date) FILTER (date >= DATE '{COVERAGE_CUTOFF}')) + 1,
344                   count(temp_max_c) FILTER (date >= DATE '{COVERAGE_CUTOFF}')
345            FROM {relation} GROUP BY city ORDER BY city
346            """
347        ).fetchall()
348    ]

351def build_manifest(
352    con: duckdb.DuckDBPyConnection,
353    relation: str,
354    sources: list[Source],
355    skipped: list[dict[str, str]],
356    started: datetime,
357) -> dict[str, Any]:
358    nachweis = reconcile(con, relation, sources)
359    herkunft = [
360        {
361            "layer": BRONZE_LAYER,
362            "dataset": NOAA_DATASET,
363            "station_id": quelle.station,
364            "city": quelle.city,
365            "run_id": quelle.run_id,
366        }
367        for quelle in sources
368    ]
369    return {
370        "status": "complete",
371        "contract_version": CONTRACT_VERSION,
372        "code_version": code_stand(),
373        "layer": SILVER_LAYER,
374        "table": TABLE,
375        "grain": "eine Stadt an einem Tag",
376        "built_at_utc": started.isoformat().replace("+00:00", "Z"),
377        "tool": f"duckdb {duckdb.__version__}",
378        "format": "parquet (snappy)",
379        "row_count": nachweis["row_count"],
380        # Wetter wird nicht raeumlich verbunden, sondern ueber Stadt und Tag.
381        "join_keys": ["city", "date"],
382        "derived_from": herkunft,
383        # Ein Abdruck ueber die Eingaben: Wer diese Tabelle benutzt, nennt ihn
384        # und sagt damit, auf welchem Stand er ruht - ein Pfad koennte das nicht.
385        "input_fingerprint": eingabe_abdruck(herkunft),
386        "skipped_stations": skipped,
387        "reconciliation": nachweis,
388        "coverage": coverage(con, relation),
389        "schema": schema_of(con, relation),
390        "notes": {
391            "units": (
392                "temp_max_c und temp_min_c in Grad Celsius, precipitation_mm "
393                "und snowfall_mm in Millimetern. Die Quelle fuehrt Temperatur "
394                "und Niederschlag in Zehnteln, Schneefall schon in Millimetern"
395            ),
396            "trace": (
397                "precipitation_trace bedeutet messbarer Niederschlag unter der "
398                "Auflösung; der Wert ist dann 0 und nicht fehlend"
399            ),
400            "missing_vs_zero": (
401                "ein leeres Quellfeld wird NULL und nicht 0 - bei Niederschlag "
402                "der entscheidende Unterschied"
403            ),
404            "coverage_before_1970": (
405                "die Reihen reichen bis 1942 zurueck und sind dort "
406                "lueckenhaft; ab dem Stichtag in coverage.since_cutoff ist "
407                "sie vollstaendig"
408            ),
409            "source_lag": (
410                "GHCN-Daily veroeffentlicht mit rund vier Tagen Nachlauf; ein "
411                "T-1-Anspruch nach Absprache 7.1 ist hier nicht erfuellbar"
412            ),
413            "extra_column": "_extra wird nicht uebernommen und bleibt in Bronze",
414        },
415    }

418def build(data_root: Path, stations: list[str]) -> Path:
419    """Baut die Tabelle und liefert das veroeffentlichte Verzeichnis."""
420    started = datetime.now(timezone.utc)
421    sources = collect_sources(data_root, stations)
422    for quelle in sources:
423        log.info("Quelle: %s (%s) %s", quelle.station, quelle.city, quelle.run_id)
424    uebergangen = skipped_stations(data_root, sources)
425    for eintrag in uebergangen:
426        log.info(
427            "uebergangen: %s (%s) - %s",
428            eintrag["station_id"],
429            eintrag["city"],
430            eintrag["reason"],
431        )
432
433    con = connect()
434    try:
435        con.execute(f"CREATE TEMP VIEW weather_silver AS {build_query(sources)}")
436        manifest = build_manifest(
437            con, "weather_silver", sources, uebergangen, started
438        )
439        ziel = publish(con, "weather_silver", table_dir(data_root), manifest)
440    finally:
441        con.close()
442    for_root(data_root).publish_table(ziel)
443    return ziel

446def main() -> int:
447    args = arguments()
448    from storage.verwaltet import sperre
449    sperre(for_root(args.data_root).local_root, worker_erlaubt=True)
450    from storage.analysesitzung import schreibzugriff
451    with schreibzugriff(for_root(args.data_root).local_root, invalidieren=True):
452        try:
453            # Ueber die Ablage und nicht ueber den Pfad: ``publish_table``
454            # entfernt im s3-Modus die lokale Arbeitskopie, und dieser
455            # Schlussbericht las sie trotzdem - jeder Lauf endete dort mit
456            # "Invalid manifest" und Exitcode 1, obwohl die Tabelle sauber
457            # publiziert war. Siehe ``Storage.tabellen_manifest``.
458            ablage = for_root(args.data_root)
459            ziel = build(ablage.local_root, args.station)
460            manifest = ablage.tabellen_manifest(ziel)
461            log.info("Silver-Tabelle geschrieben: %s", ziel)
462            log.info("  Zeilen: %s", f"{manifest['row_count']:,}")
463            for abdeckung in manifest["coverage"]:
464                log.info(
465                    "  %s (%s): %s bis %s, %s Zeilen, %s fehlende Tage, "
466                    "%s Tage mit Spurenmenge, %s beanstandet",
467                    abdeckung["city"],
468                    abdeckung["station_id"],
469                    abdeckung["first_date"],
470                    abdeckung["last_date"],
471                    f"{abdeckung['row_count']:,}",
472                    abdeckung["missing_days"],
473                    f"{abdeckung['trace_precipitation_days']:,}",
474                    abdeckung["suspect_rows"],
475                )

def build_query(sources: list[Source]) -> str: 
213def build_query(sources: list[Source]) -> str:
214    """Die Abfrage, die aus Stationsreihen Stadt-Tage macht."""
215    lesen = " UNION ALL ".join(
216        f"SELECT '{quelle.city}' AS city, '{quelle.station}' AS station_id, * "
217        f"FROM read_parquet('{quelle.parquet}')"
218        for quelle in sources
219    )
220    return f"""
221    WITH roh AS ({lesen})
222    SELECT
223        city,
224        CAST(DATE AS DATE)                          AS date,
225        station_id,
226        NAME                                        AS station_name,
227        {measurement_sql('TMAX', tenths=True)}      AS temp_max_c,
228        {measurement_sql('TMIN', tenths=True)}      AS temp_min_c,
229        {measurement_sql('PRCP', tenths=True)}      AS precipitation_mm,
230        {measurement_sql('SNOW', tenths=False)}     AS snowfall_mm,
231        -- Spurenmenge: messbar vorhanden, aber unter der Auflösung. Der Wert
232        -- steht als 0 da, was fachlich etwas anderes ist als trocken.
233        coalesce({flag_sql('PRCP', 1)} = '{TRACE_FLAG}', FALSE) AS precipitation_trace,
234        coalesce({flag_sql('SNOW', 1)} = '{TRACE_FLAG}', FALSE) AS snowfall_trace,
235        list_filter([
236            CASE WHEN TRY_CAST(TMAX AS DOUBLE) IS NULL
237                   OR TRY_CAST(TMIN AS DOUBLE) IS NULL
238                 THEN 'MISSING_TEMPERATURE' END,
239            CASE WHEN TRY_CAST(PRCP AS DOUBLE) IS NULL
240                 THEN 'MISSING_PRECIPITATION' END,
241            CASE WHEN TRY_CAST(SNOW AS DOUBLE) IS NULL
242                 THEN 'MISSING_SNOWFALL' END,
243            -- Zweite Position von M,Q,S ist NOAAs Beanstandung. Fast immer
244            -- leer, aber nicht immer.
245            CASE WHEN {flag_sql('TMAX', 2)} IS NOT NULL THEN 'SUSPECT_TEMP_MAX' END,
246            CASE WHEN {flag_sql('TMIN', 2)} IS NOT NULL THEN 'SUSPECT_TEMP_MIN' END,
247            CASE WHEN {flag_sql('PRCP', 2)} IS NOT NULL THEN 'SUSPECT_PRECIPITATION' END,

250    FROM roh
251    ORDER BY city, date
252    """

255def reconcile(

258    """Weist nach, dass die Granularität haelt und keine Zeile verloren ging."""
259    quellzeilen = sum(
260        con.execute(

263        for quelle in sources

288        "bronze_rows": quellzeilen,
