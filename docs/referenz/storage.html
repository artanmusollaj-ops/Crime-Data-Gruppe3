storage

Die Datenablage: lokal, SeaweedFS ueber S3, oder hybrid.

Hybrid READ:    lokal zuerst, S3 als Rueckfall
Hybrid WRITE:   lokal, pruefen, dann gezielt nach S3 publizieren
Serving DuckDB: immer lokal

Der Modus kommt aus STORAGE_BACKEND (local ist die Vorgabe), die uebrige Konfiguration aus der Umgebung oder .env - siehe config. Pipelines holen sich die Ablage mit get_storage() beziehungsweise for_root(data_root), wenn eine --data-root uebergeben wurde.

Die kleine API fuer Notebooks und Skripte (an der Standardablage)::

from storage import resolve_read_path, exists, read_text
pfad = resolve_read_path("silver", "dataset=crime_offense", "part-0001.parquet")

575        """Laedt lokale Dateien nach S3, ``manifest.json`` zuletzt.
576
577        Gibt zurueck, was jetzt oben liegt. Ohne S3 im Backend passiert
578        nichts; im Hybridmodus mit ``HYBRID_PUBLISH_AFTER_WRITE=false`` auch
579        nicht - dann steht es im Log.
580        """
581        liste = [paths.relativ(d) for d in dateien]
582        if not liste:
583            return []
584        if not self.config.publiziert:
585            if self.backend == "hybrid":
586                LOG.info("Not published (HYBRID_PUBLISH_AFTER_WRITE=false): %d file(s)", len(liste))
587            return []
588
589        reihenfolge = sorted(liste, key=lambda r: (r.rsplit("/", 1)[-1] == MANIFEST, r))
590        oben: list[str] = []
591        for rel in reihenfolge:
592            lokal = self.local_path(rel)
593            daten = lokal.read_bytes()
594            md5 = hashlib.md5(daten).hexdigest()
595
596            vorhanden = self.client.head(rel)
597            if vorhanden is not None:
598                gleich = vorhanden.size == len(daten) and (
599                    etag_md5(vorhanden.etag) in (None, md5)
600                )
601                if gleich:
602                    LOG.info("Already published, identical: %s", rel)
603                    oben.append(rel)
604                    continue
605                if not ueberschreiben:
606                    raise PublishError(
607                        f"{self.s3_path(rel)} existiert bereits mit anderem Inhalt "
608                        f"({vorhanden.size} Bytes, lokal {len(daten)}). Laeufe sind "
609                        "unveraenderlich; nichts wird still ueberschrieben."
610                    )
611                LOG.info("Replacing published table file: %s", rel)
612
613            try:
614                self.client.put(rel, daten)
615            except Exception as error:  # noqa: BLE001 - jede Transportstoerung ist hier gleich
616                raise PublishError(f"Upload fehlgeschlagen: {self.s3_path(rel)}: {error}") from error
617
618            nachher = self.client.head(rel)
619            if nachher is None or nachher.size != len(daten):
620                raise PublishError(
621                    f"Upload nicht bestaetigt: {self.s3_path(rel)} "
622                    f"({'fehlt' if nachher is None else f'{nachher.size} statt {len(daten)} Bytes'})"
623                )
624            if etag_md5(nachher.etag) not in (None, md5):
625                raise PublishError(f"Pruefsumme nach Upload weicht ab: {self.s3_path(rel)}")
626            LOG.info("Published: %s", rel)
627            oben.append(rel)
628        return oben

574    def publish_files(self, dateien: Iterable[str], *, ueberschreiben: bool = False) -> list[str]:
575        """Laedt lokale Dateien nach S3, ``manifest.json`` zuletzt.
576
577        Gibt zurueck, was jetzt oben liegt. Ohne S3 im Backend passiert
578        nichts; im Hybridmodus mit ``HYBRID_PUBLISH_AFTER_WRITE=false`` auch
579        nicht - dann steht es im Log.
580        """
581        liste = [paths.relativ(d) for d in dateien]
582        if not liste:
583            return []
584        if not self.config.publiziert:
585            if self.backend == "hybrid":
586                LOG.info("Not published (HYBRID_PUBLISH_AFTER_WRITE=false): %d file(s)", len(liste))
587            return []
588
589        reihenfolge = sorted(liste, key=lambda r: (r.rsplit("/", 1)[-1] == MANIFEST, r))
590        oben: list[str] = []
591        for rel in reihenfolge:
592            lokal = self.local_path(rel)
593            daten = lokal.read_bytes()
594            md5 = hashlib.md5(daten).hexdigest()
595
596            vorhanden = self.client.head(rel)
597            if vorhanden is not None:
598                gleich = vorhanden.size == len(daten) and (
599                    etag_md5(vorhanden.etag) in (None, md5)
600                )
601                if gleich:
602                    LOG.info("Already published, identical: %s", rel)
603                    oben.append(rel)
604                    continue
605                if not ueberschreiben:
606                    raise PublishError(
607                        f"{self.s3_path(rel)} existiert bereits mit anderem Inhalt "
608                        f"({vorhanden.size} Bytes, lokal {len(daten)}). Laeufe sind "
609                        "unveraenderlich; nichts wird still ueberschrieben."
610                    )
611                LOG.info("Replacing published table file: %s", rel)
612
613            try:
614                self.client.put(rel, daten)
615            except Exception as error:  # noqa: BLE001 - jede Transportstoerung ist hier gleich
616                raise PublishError(f"Upload fehlgeschlagen: {self.s3_path(rel)}: {error}") from error
617
618            nachher = self.client.head(rel)
619            if nachher is None or nachher.size != len(daten):
620                raise PublishError(
621                    f"Upload nicht bestaetigt: {self.s3_path(rel)} "
622                    f"({'fehlt' if nachher is None else f'{nachher.size} statt {len(daten)} Bytes'})"
623                )
624            if etag_md5(nachher.etag) not in (None, md5):
625                raise PublishError(f"Pruefsumme nach Upload weicht ab: {self.s3_path(rel)}")
626            LOG.info("Published: %s", rel)
627            oben.append(rel)
628        return oben
