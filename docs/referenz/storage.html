storage

Die Datenablage: lokal, SeaweedFS ueber S3, oder hybrid.

Hybrid READ:    lokal zuerst, S3 als Rueckfall
Hybrid WRITE:   lokal, pruefen, dann gezielt nach S3 publizieren
Serving DuckDB: immer lokal

Der Modus kommt aus STORAGE_BACKEND (local ist die Vorgabe), die uebrige Konfiguration aus der Umgebung oder .env - siehe config. Pipelines holen sich die Ablage mit get_storage() beziehungsweise for_root(data_root), wenn eine --data-root uebergeben wurde.

Die kleine API fuer Notebooks und Skripte (an der Standardablage)::

from storage import resolve_read_path, exists, read_text
pfad = resolve_read_path("silver", "dataset=crime_offense", "part-0001storage.parquet")

208    def _mit_wiederholung(self, was: str, aufruf):
209        """Fuehrt ``aufruf()`` aus; voruebergehende Fehler werden wiederholt."""
210        from botocore.exceptions import BotoCoreError, ClientError
211
212        for versuch in range(VERSUCHE):
213            try:
214                return aufruf()
215            except ClientError as fehler:
216                code = str(fehler.response.get("Error", {}).get("Code", ""))
217                if code in ("404", "NoSuchKey", "NotFound"):
218                    raise
219                letzter = fehler
220                if code not in VORUEBERGEHEND:
221                    break
222            except BotoCoreError as fehler:
223                letzter = fehler
224            if versuch < VERSUCHE - 1:
225                LOG.warning("SeaweedFS: %s bei %s - Versuch %d von %d, Pause %.0f s",
226                            fehlerkennung(letzter), was, versuch + 1, VERSUCHE, PAUSEN[versuch])
227                time.sleep(PAUSEN[versuch])
228        raise self._uebersetzt(letzter, was) from letzter

230    def _uebersetzt(self, fehler: Exception, was: str) -> StorageError:
231        """Ein boto-Fehler als ein Satz - Endpoint und Ursache, keine Schluessel."""
232        from botocore.exceptions import ClientError, EndpointConnectionError
233
234        if isinstance(fehler, EndpointConnectionError):
235            return StorageError(
236                f"SeaweedFS nicht erreichbar: {self.endpoint_url} ({was}). "
237                "Stimmt S3_ENDPOINT, und loest der Hostname auf? "
238                "'python -m storage' prueft die Verbindung."
239            )
240        if isinstance(fehler, ClientError):
241            code = str(fehler.response.get("Error", {}).get("Code", "?"))
242            return StorageError(
243                f"SeaweedFS lehnt ab: {code} bei {was} auf {self.endpoint_url}, "
244                f"Bucket {self.bucket!r}. Bei AccessDenied/InvalidAccessKeyId/"
245                "SignatureDoesNotMatch die Schluessel pruefen, bei NoSuchBucket den Bucket."
246            )
247        return StorageError(f"SeaweedFS: {type(fehler).__name__} bei {was}: {fehler}")

277    def download(self, key: str, ziel: Path) -> None:
278        """Schreibt ein Objekt in eine Datei, ohne es ganz im Speicher zu halten.
279
280        ``get`` liest den ganzen Koerper - fuer einen Bronze-Snapshot von
281        mehreren GiB zu viel neben einem Silver-Bau. Ein Faden und eine
282        Verbindung halten den Speicher klein; schneller muss es nicht sein.
283        """
284        from boto3.s3.transfer import TransferConfig
285        from botocore.exceptions import ClientError
286
287        transfer = TransferConfig(max_concurrency=1, use_threads=False)
288        try:
289            self._mit_wiederholung(
290                f"DOWNLOAD {key}",
291                lambda: self._client.download_file(self.bucket, key, str(ziel),
292                                                   Config=transfer))
293        except ClientError as fehler:
294            raise self._uebersetzt(fehler, f"DOWNLOAD {key}") from fehler

296    def list(self, prefix: str) -> list[Objekt]:
297        def alle() -> list[Objekt]:
298            gefunden: list[Objekt] = []
299            paginator = self._client.get_paginator("list_objects_v2")
300            for seite in paginator.paginate(Bucket=self.bucket, Prefix=prefix):
301                for eintrag in seite.get("Contents", []):
302                    gefunden.append(Objekt(eintrag["Key"], int(eintrag["Size"]),
303                                           eintrag.get("ETag", "")))
304            return gefunden
305
306        return self._mit_wiederholung(f"LIST {prefix}", alle)

308    def delete(self, key: str) -> None:
309        # S3 antwortet auf DELETE eines fehlenden Schluessels mit 204, nicht
310        # mit 404 - loeschen ist damit wiederholbar. Das traegt
311        # ``Storage.entferne_oben``: Ein Abbruch mitten im Praefix laesst sich
312        # durch einen zweiten Lauf zu Ende bringen.
313        #
314        # **In einem versionierten Eimer ist das kein Freigeben.** Es setzt
315        # eine Loeschmarkierung; die Fassung darunter bleibt und belegt
316        # weiter Platz. Wer Platz will, braucht ``entferne_fassung``.
317        self._mit_wiederholung(
318            f"DELETE {key}",
319            lambda: self._client.delete_object(Bucket=self.bucket, Key=key))
320
321    def fassungen(self, prefix: str) -> list[Fassung]:
322        """Jede Fassung unter ``prefix``, alte und Markierungen inbegriffen.
323
324        ``list_object_versions`` statt ``list_objects_v2``: Das zweite nennt
325        nur die jeweils aktuelle Fassung, und genau deshalb war der Rest
326        unsichtbar.
327        """
328        def alle() -> list[Fassung]:
329            gefunden: list[Fassung] = []
330            paginator = self._client.get_paginator("list_object_versions")
331            for seite in paginator.paginate(Bucket=self.bucket, Prefix=prefix):
332                for e in seite.get("Versions", []):
333                    gefunden.append(Fassung(
334                        e["Key"], str(e.get("VersionId") or ""),
335                        int(e.get("Size") or 0), bool(e.get("IsLatest")), False))
336                for e in seite.get("DeleteMarkers", []):
337                    gefunden.append(Fassung(
338                        e["Key"], str(e.get("VersionId") or ""),
339                        0, bool(e.get("IsLatest")), True))
340            return gefunden
341
342        return self._mit_wiederholung(f"VERSIONS {prefix}", alle)
343
344    def entferne_fassung(self, key: str, version: str) -> None:
345        """Eine einzelne Fassung endgueltig - das Einzige, was Platz freigibt."""
346        self._mit_wiederholung(
347            f"DELETE {key}?{version}",
348            lambda: self._client.delete_object(
349                Bucket=self.bucket, Key=key, VersionId=version))

 753    def list_files(self, prefix: str | Path, muster: str = "**/*") -> list[str]:
 754        """Dateien unter ``prefix``, als relative Ablagepfade, sortiert.

 756        ``muster`` gilt relativ zu ``prefix`` und wie bei ``Path.glob``:
 757        ``**/`` deckt auch null Verzeichnisebenen ab. Versteckte Teile
 758        (``.part``, ``.partial``) zaehlen nie. hybrid: lokal und S3 vereint,

 764            wurzel = self.local_path(praefix)

 766                for pfad in wurzel.glob(muster):
 767                    if pfad.is_file():

 770                            gefunden.add(paths.relativ(praefix, rel))

 772            for objekt in self.client.list(praefix + "/" if praefix else ""):
 773                rel = objekt.key[len(praefix) + 1:] if praefix else objekt.key
 774                if paths.versteckt(rel) or not passt(rel, muster):
 775                    continue
 776                gefunden.add(paths.relativ(praefix, rel))
 777        return sorted(gefunden)

 779    def groessen_unter(self, prefix: str | Path, muster: str = "**/*", *,
 780                       lokal: bool = False) -> dict[str, int]:
 781        """Jede Datei unter ``prefix`` mit ihrer Groesse in Bytes.
 782
 783        Dieselbe Auswahl wie ``list_files`` und darum dieselben Regeln:
 784        Versteckte Teile (``.part``, ``.partial``) zaehlen nie, und im
 785        Hybridmodus gewinnt die lokale Datei - dieselbe Datei wird **einmal**
 786        gezaehlt, nicht zweimal. Gemessen wird lokal mit ``st_size``, auf S3
 787        mit ``Objekt.size``; S3 wird zuerst gelesen, damit die lokale Groesse
 788        sie ueberschreibt.

 790        ``lokal=True`` misst nur die Arbeitskopie auf dieser Platte, auch im
 791        ``s3``-Modus, in dem die Ablage sonst gar nicht lokal liest. Das ist
 792        die Frage, die ``aufraeumen`` stellt (was gibt das Loeschen frei?) und
 793        die ``python -m storage publizieren`` stellt (was ist noch
 794        hochzuladen?) - beide meinen Bytes auf der Platte, nicht den
 795        gemeinsamen Bestand.
 796        """
 797        praefix = paths.relativ(prefix)
 798        gefunden: dict[str, int] = {}
 799        if self.config.remote and not lokal:
 800            for objekt in self.client.list(praefix + "/" if praefix else ""):
 801                rel = objekt.key[len(praefix) + 1:] if praefix else objekt.key
 802                if paths.versteckt(rel) or not passt(rel, muster):
 803                    continue
 804                gefunden[paths.relativ(praefix, rel)] = objekt.size
 805        if self.config.liest_lokal or lokal:
 806            wurzel = self.local_path(praefix)
 807            if wurzel.exists():
 808                for pfad in wurzel.glob(muster):
 809                    if not pfad.is_file():
 810                        continue
 811                    rel = pfad.relative_to(wurzel).as_posix()
 812                    if not paths.versteckt(rel):
 813                        gefunden[paths.relativ(praefix, rel)] = pfad.stat().st_size
 814        return dict(sorted(gefunden.items()))

 910    def publish_files(self, dateien: Iterable[str], *, ueberschreiben: bool = False) -> list[str]:
 911        """Laedt lokale Dateien nach S3, ``manifest.json`` zuletzt.
 912
 913        Gibt zurueck, was jetzt oben liegt. Ohne S3 im Backend passiert
 914        nichts; im Hybridmodus mit ``HYBRID_PUBLISH_AFTER_WRITE=false`` auch
 915        nicht - dann steht es im Log.
 916
 917        Was oben schon liegt, wird **einmal je Praefix** aufgelistet, nicht je
 918        Datei mit ``HEAD`` erfragt (siehe ``_bestand_oben``). Jede Entscheidung
 919        bleibt dieselbe: identisch oben wird uebersprungen, abweichend oben ist
 920        ohne ``ueberschreiben`` ein Fehler, und ``_ist_identisch`` entscheidet
 921        allein, was "gleich" heisst.
 922        """
 923        liste = [paths.relativ(d) for d in dateien]
 924        if not liste:
 925            return []
 926        if not self.config.publiziert:
 927            if self.backend == "hybrid":
 928                LOG.info("Not published (HYBRID_PUBLISH_AFTER_WRITE=false): %d file(s)", len(liste))
 929            return []
 930
 931        reihenfolge = sorted(liste, key=lambda r: (r.rsplit("/", 1)[-1] == MANIFEST, r))
 932        bestand = self._bestand_oben(reihenfolge)
 933        oben: list[str] = []
 934        for rel in reihenfolge:
 935            lokal = self.local_path(rel)
 936            daten = lokal.read_bytes()
 937            md5 = hashlib.md5(daten).hexdigest()
 938
 939            vorhanden = bestand.get(rel)
 940            if vorhanden is not None:
 941                if self._ist_identisch(rel, vorhanden, daten, md5):
 942                    LOG.info("Already published, identical: %s", rel)
 943                    oben.append(rel)
 944                    continue
 945                if not ueberschreiben:
 946                    raise PublishError(
 947                        f"{self.s3_path(rel)} existiert bereits mit anderem Inhalt "
 948                        f"({vorhanden.size} Bytes, lokal {len(daten)}). Laeufe sind "
 949                        "unveraenderlich; nichts wird still ueberschrieben."
 950                    )
 951                LOG.info("Replacing published table file: %s", rel)
 952
 953            try:
 954                self.client.put(rel, daten)
 955            except Exception as error:  # noqa: BLE001 - jede Transportstoerung ist hier gleich
 956                raise PublishError(f"Upload fehlgeschlagen: {self.s3_path(rel)}: {error}") from error
 957
 958            nachher = self.client.head(rel)
 959            if nachher is None or nachher.size != len(daten):
 960                raise PublishError(
 961                    f"Upload nicht bestaetigt: {self.s3_path(rel)} "
 962                    f"({'fehlt' if nachher is None else f'{nachher.size} statt {len(daten)} Bytes'})"
 963                )
 964            if not self._ist_identisch(rel, nachher, daten, md5):
 965                raise PublishError(f"Pruefsumme nach Upload weicht ab: {self.s3_path(rel)}")
 966            LOG.info("Published: %s", rel)
 967            oben.append(rel)
 968        return oben

 970    def _bestand_oben(self, dateien: list[str]) -> dict[str, Objekt]:
 971        """Was von ``dateien`` schon oben liegt - **ein** ``list`` je Praefix.
 972
 973        Bis zum 5.10.2026 fragte ``publish_files`` je Datei mit ``HEAD``, ob
 974        oben schon etwas liegt. Beim Publizieren der sechs Jahrgaenge waren das
 975        446 Anfragen in Folge, und der Endpunkt drosselte 350 davon: Unter
 976        dichter Anfragefolge antwortet er auf einen fehlenden Schluessel mit
 977        403 statt 404. Gemessen am 5.10.2026 - 120 Anfragen in 8,5 s ergaben
 978        120 x 403, derselbe fehlende Schluessel einzeln abgefragt 403 einmal
 979        und 404 viermal. ``403`` steht in ``VORUEBERGEHEND``, also holte
 980        ``_mit_wiederholung`` nach einer Sekunde das ehrliche 404 - das
 981        Ergebnis stimmte, der Upload dauerte doppelt so lang.
 982
 983        Ein ``list`` beantwortet dieselbe Frage fuer alle Dateien eines
 984        Verzeichnisses: ``Objekt`` traegt ``key``, ``size`` und ``etag``, und
 985        genau das - und nichts darueber hinaus - braucht ``_ist_identisch``.
 986        Die Zahl der Anfragen fuer die Vorabpruefung faellt damit von N auf
 987        eins je Verzeichnis.
 988
 989        Aufgelistet wird je **Praefix**, nicht einmal fuer alles: Ein Release
 990        traegt Dateien unter ``gold/dataset=...`` und ``serving/dataset=...``,
 991        und ein ``list`` auf den gemeinsamen Stamm brachte jedes andere
 992        Release mit.
 993        """
 994        gesucht: dict[str, set[str]] = {}
 995        for rel in dateien:
 996            verzeichnis = rel.rpartition("/")[0]
 997            # Ein Schluessel ohne Verzeichnis (``write_text`` kann einen
 998            # anlegen) wird mit sich selbst als Praefix aufgelistet: ``list("")``
 999            # waere der ganze Bucket. Nachgeschlagen wird ohnehin exakt.
1000            gesucht.setdefault(verzeichnis + "/" if verzeichnis else rel, set()).add(rel)
1001        bestand: dict[str, Objekt] = {}
1002        for praefix, schluessel in gesucht.items():
1003            for objekt in self.client.list(praefix):
1004                if objekt.key in schluessel:
1005                    bestand[objekt.key] = objekt
1006        return bestand

1008    def _ist_identisch(self, rel: str, oben: Objekt, daten: bytes, md5: str) -> bool:
1009        """Ob das Objekt oben byteweise dasselbe ist wie ``daten``.
1010
1011        **Gleiche Groesse ist kein Beweis.** Genau das stand hier bis zum
1012        15.9.2026: Liess sich das ETag nicht als MD5 lesen, genuegte die
1013        Byteanzahl - und drei Bytes ``NEU`` galten als identisch mit drei
1014        Bytes ``ALT``. Der Upload unterblieb, und die Zusage unveraenderlicher
1015        Laeufe ruhte auf einem Zufall.
1016
1017        Ein ETag ist nicht immer eine Pruefsumme: Bei mehrteiligen Uploads
1018        setzt S3 einen Hash ueber die Teil-Hashes mit angehaengtem ``-N``.
1019        Dann bleibt nur, den Inhalt zu holen und zu vergleichen. Das kostet
1020        einen Download - aber nur dann, wenn die Groessen ohnehin gleich sind
1021        und die Pruefsumme nichts hergibt.
1022        """
1023        if oben.size != len(daten):
1024            return False
1025        bekannt = etag_md5(oben.etag)
1026        if bekannt is not None:
1027            return bekannt == md5
1028        LOG.info("ETag ist keine Pruefsumme (%s), Inhalt wird geholt: %s",
1029                 oben.etag, self.s3_path(rel))
1030        try:
1031            return self.client.get(rel) == daten
1032        except Exception as fehler:  # noqa: BLE001 - jede Transportstoerung ist hier gleich
1033            raise PublishError(
1034                f"Inhalt nicht pruefbar: {self.s3_path(rel)}: {fehler}. Das ETag "
1035                "ist keine Pruefsumme, und der Vergleichsdownload scheiterte - "
1036                "gleiche Groesse allein gilt nicht als gleicher Inhalt."
1037            ) from fehler

1178    def hole(self, rel: str | Path) -> Path:
1179        """Ein Objekt von S3 in die lokale Wurzel - geprueft und unteilbar.
1180
1181        Liegt es lokal schon identisch, passiert nichts. Geschrieben wird in
1182        eine ``.part``-Datei daneben, verglichen (Groesse, dann MD5 gegen das
1183        ETag, wo der Dienst eines liefert), dann umbenannt.
1184        """
1185        rel = paths.relativ(rel)
1186        if not self.config.remote:
1187            raise PublishError(f"STORAGE_BACKEND={self.backend} kennt kein S3 - nichts zu holen")
1188        oben = self.client.head(rel)
1189        if oben is None:
1190            raise StorageError(f"nicht auf SeaweedFS: {self.s3_path(rel)}")
1191        ziel = self.local_path(rel)
1192        if ziel.is_file() and ziel.stat().st_size == oben.size:
1193            bekannt = etag_md5(oben.etag)
1194            if bekannt is None or bekannt == hashlib.md5(ziel.read_bytes()).hexdigest():
1195                LOG.debug("Already local, identical: %s", rel)
1196                return ziel
1197        try:
1198            daten = self.client.get(rel)
1199        except Exception as fehler:  # noqa: BLE001 - jede Transportstoerung ist hier gleich
1200            raise StorageError(f"Download fehlgeschlagen: {self.s3_path(rel)}: {fehler}") from fehler
1201        if len(daten) != oben.size:
1202            raise StorageError(
1203                f"Download unvollstaendig: {self.s3_path(rel)} "
1204                f"({len(daten)} statt {oben.size} Bytes)"
1205            )
1206        bekannt = etag_md5(oben.etag)
1207        if bekannt is not None and bekannt != hashlib.md5(daten).hexdigest():
1208            raise StorageError(f"Pruefsumme nach Download weicht ab: {self.s3_path(rel)}")
1209        ziel.parent.mkdir(parents=True, exist_ok=True)
1210        arbeit = ziel.with_name(f".{ziel.name}.{id(self) & 0xffff:x}.part")
1211        arbeit.write_bytes(daten)
1212        arbeit.replace(ziel)
1213        LOG.info("Fetched: %s", rel)
1214        return ziel

1269

1302            )

1308        for key in schluessel:
1309            self.client.delete(key)
1310            geloescht.append(key)
1311            LOG.info("Removed from S3: %s", key)
