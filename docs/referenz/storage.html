storage

Die Datenablage: lokal, SeaweedFS ueber S3, oder hybrid.

Hybrid READ:    lokal zuerst, S3 als Rueckfall
Hybrid WRITE:   lokal, pruefen, dann gezielt nach S3 publizieren
Serving DuckDB: immer lokal

Der Modus kommt aus STORAGE_BACKEND (local ist die Vorgabe), die uebrige Konfiguration aus der Umgebung oder .env - siehe config. Pipelines holen sich die Ablage mit get_storage() beziehungsweise for_root(data_root), wenn eine --data-root uebergeben wurde.

Die kleine API fuer Notebooks und Skripte (an der Standardablage)::

from storage import resolve_read_path, exists, read_text
pfad = resolve_read_path("silver", "dataset=crime_offense", "part-0001.parquet")

160    def _uebersetzt(self, fehler: Exception, was: str) -> StorageError:
161        """Ein boto-Fehler als ein Satz - Endpoint und Ursache, keine Schluessel."""
162        from botocore.exceptions import ClientError, EndpointConnectionError
163
164        if isinstance(fehler, EndpointConnectionError):
165            return StorageError(
166                f"SeaweedFS nicht erreichbar: {self.endpoint_url} ({was}). "
167                "Stimmt S3_ENDPOINT, und loest der Hostname auf? "
168                "'python -m storage' prueft die Verbindung."
169            )
170        if isinstance(fehler, ClientError):
171            code = str(fehler.response.get("Error", {}).get("Code", "?"))
172            return StorageError(
173                f"SeaweedFS lehnt ab: {code} bei {was} auf {self.endpoint_url}, "
174                f"Bucket {self.bucket!r}. Bei AccessDenied/InvalidAccessKeyId/"
175                "SignatureDoesNotMatch die Schluessel pruefen, bei NoSuchBucket den Bucket."
176            )
177        return StorageError(f"SeaweedFS: {type(fehler).__name__} bei {was}: {fehler}")

491    def list_files(self, prefix: str | Path, muster: str = "**/*") -> list[str]:
492        """Dateien unter ``prefix``, als relative Ablagepfade, sortiert.
493
494        ``muster`` gilt relativ zu ``prefix`` und wie bei ``Path.glob``:
495        ``**/`` deckt auch null Verzeichnisebenen ab. Versteckte Teile
496        (``.part``, ``.partial``) zaehlen nie. hybrid: lokal und S3 vereint,
497        lokal gewinnt bei gleichem Pfad.
498        """
499        praefix = paths.relativ(prefix)
500        gefunden: set[str] = set()
501        if self.config.liest_lokal:
502            wurzel = self.local_path(praefix)
503            if wurzel.exists():
504                for pfad in wurzel.glob(muster):
505                    if pfad.is_file():
506                        rel = pfad.relative_to(wurzel).as_posix()
507                        if not paths.versteckt(rel):
508                            gefunden.add(paths.relativ(praefix, rel))
509        if self.config.remote:
510            for objekt in self.client.list(praefix + "/" if praefix else ""):
511                rel = objekt.key[len(praefix) + 1:] if praefix else objekt.key
512                if paths.versteckt(rel) or not passt(rel, muster):
513                    continue
514                gefunden.add(paths.relativ(praefix, rel))
515        return sorted(gefunden)

517    def laeufe(self, prefix: str | Path, muster: str = f"**/run_*/{MANIFEST}", *,
518               status: frozenset[str] = FERTIG, streng: bool = False,
519               fehler: type[Exception] = PublishError) -> list[Lauf]:
520        """Fertige Laeufe unter ``prefix`` - lokal, auf S3 oder beides.
521
522        Jeder ``Lauf`` traegt seinen relativen Ablagepfad in ``relativ``;
523        ``verzeichnis`` ist der lokale Pfad, der bei einem S3-Lauf nicht
524        existieren muss. Ein Lauf, dessen Manifest nicht fertig sagt, gilt
525        nicht - auch nicht, wenn er lokal liegt und S3 einen fertigen hat.
526        """
527        praefix = paths.relativ(prefix)
528        gefunden: dict[str, Lauf] = {}
529        if self.config.liest_lokal:
530            wurzel = self.local_path(praefix)
531            for lauf in fertige_laeufe(wurzel, muster, streng=streng, fehler=fehler,
532                                       status=status):
533                rel = paths.relativ(praefix, lauf.verzeichnis.relative_to(wurzel).as_posix())
534                gefunden[rel] = Lauf(lauf.verzeichnis, lauf.manifest, rel)
535        if self.config.remote:
536            for objekt in self.client.list(praefix + "/" if praefix else ""):
537                if not objekt.key.endswith("/" + MANIFEST):
538                    continue
539                rel = objekt.key[len(praefix) + 1:] if praefix else objekt.key
540                if paths.versteckt(rel) or not passt(rel, muster):
541                    continue
542                rel_dir = objekt.key[: -len("/" + MANIFEST)]
543                if rel_dir in gefunden:
544                    continue
545                try:
546                    manifest = json.loads(self.client.get(objekt.key).decode("utf-8"))
547                except (OSError, ValueError) as error:
548                    raise fehler(f"Invalid manifest: {self.s3_path(objekt.key)}") from error
549                if str(manifest.get("status", "")).lower() not in status:
550                    LOG.warning("Lauf uebersprungen, nicht fertig: %s", self.s3_path(rel_dir))
551                    continue
552                gefunden[rel_dir] = Lauf(self.local_path(rel_dir), manifest, rel_dir)
553        return [gefunden[k] for k in sorted(gefunden)]

555    def tabellen(self, schichten: tuple[str, ...] = ("silver", "gold")) -> list[Tabelle]:
556        """Fertige Silver- und Gold-Tabellen, mit relativem Ablagepfad.
557
558        *Fertig* heisst hier dasselbe wie bei ``laeufe``: Das Manifest sagt es.
559        Bis zum 15.9.2026 galt jede Tabelle mit Manifest als gebaut, egal was
560        darin stand - eine Tabelle mitten im Publizieren also auch. Damit half
561        der Status niemandem: Er wurde gesetzt und nie gelesen.
562        """
563        gefunden: dict[str, Tabelle] = {}
564        if self.config.liest_lokal:
565            for t in tabellen(self.local_root, schichten):
566                rel = self.relativ(t.verzeichnis) or t.verzeichnis.as_posix()
567                gefunden[rel] = Tabelle(t.schicht, t.name, t.verzeichnis, t.manifest, rel)
568        if self.config.remote:
569            for schicht in schichten:
570                for objekt in self.client.list(schicht + "/dataset="):
571                    if not objekt.key.endswith("/" + MANIFEST):
572                        continue
573                    rel_dir = objekt.key[: -len("/" + MANIFEST)]
574                    if rel_dir in gefunden or paths.versteckt(rel_dir):
575                        continue
576                    name = next(t for t in rel_dir.split("/")
577                                if t.startswith("dataset=")).removeprefix("dataset=")
578                    manifest = json.loads(self.client.get(objekt.key).decode("utf-8"))
579                    gefunden[rel_dir] = Tabelle(schicht, name, self.local_path(rel_dir),
580                                                manifest, rel_dir)
581        fertig = {}
582        for rel, tabelle in gefunden.items():
583            status = str((tabelle.manifest or {}).get("status", "")).lower()
584            if status in FERTIG:

592    def publish_files(self, dateien: Iterable[str], *, ueberschreiben: bool = False) -> list[str]:
593        """Laedt lokale Dateien nach S3, ``manifest.json`` zuletzt.
594
595        Gibt zurueck, was jetzt oben liegt. Ohne S3 im Backend passiert
596        nichts; im Hybridmodus mit ``HYBRID_PUBLISH_AFTER_WRITE=false`` auch
597        nicht - dann steht es im Log.
598        """
599        liste = [paths.relativ(d) for d in dateien]
600        if not liste:
601            return []
602        if not self.config.publiziert:
603            if self.backend == "hybrid":
604                LOG.info("Not published (HYBRID_PUBLISH_AFTER_WRITE=false): %d file(s)", len(liste))
605            return []
606
607        reihenfolge = sorted(liste, key=lambda r: (r.rsplit("/", 1)[-1] == MANIFEST, r))
608        oben: list[str] = []
609        for rel in reihenfolge:
610            lokal = self.local_path(rel)
611            daten = lokal.read_bytes()
612            md5 = hashlib.md5(daten).hexdigest()
613
614            vorhanden = self.client.head(rel)
615            if vorhanden is not None:
616                if self._ist_identisch(rel, vorhanden, daten, md5):
617                    LOG.info("Already published, identical: %s", rel)
618                    oben.append(rel)
619                    continue
620                if not ueberschreiben:
621                    raise PublishError(
622                        f"{self.s3_path(rel)} existiert bereits mit anderem Inhalt "
623                        f"({vorhanden.size} Bytes, lokal {len(daten)}). Laeufe sind "
624                        "unveraenderlich; nichts wird still ueberschrieben."
625                    )
626                LOG.info("Replacing published table file: %s", rel)
627
628            try:
629                self.client.put(rel, daten)
630            except Exception as error:  # noqa: BLE001 - jede Transportstoerung ist hier gleich
631                raise PublishError(f"Upload fehlgeschlagen: {self.s3_path(rel)}: {error}") from error
632
633            nachher = self.client.head(rel)
634            if nachher is None or nachher.size != len(daten):
635                raise PublishError(
636                    f"Upload nicht bestaetigt: {self.s3_path(rel)} "
637                    f"({'fehlt' if nachher is None else f'{nachher.size} statt {len(daten)} Bytes'})"
638                )
639            if not self._ist_identisch(rel, nachher, daten, md5):
640                raise PublishError(f"Pruefsumme nach Upload weicht ab: {self.s3_path(rel)}")
641            LOG.info("Published: %s", rel)
642            oben.append(rel)
643        return oben

645    def _ist_identisch(self, rel: str, oben: Objekt, daten: bytes, md5: str) -> bool:
646        """Ob das Objekt oben byteweise dasselbe ist wie ``daten``.
647
648        **Gleiche Groesse ist kein Beweis.** Genau das stand hier bis zum
649        15.9.2026: Liess sich das ETag nicht als MD5 lesen, genuegte die
650        Byteanzahl - und drei Bytes ``NEU`` galten als identisch mit drei
651        Bytes ``ALT``. Der Upload unterblieb, und die Zusage unveraenderlicher
652        Laeufe ruhte auf einem Zufall.
653
654        Ein ETag ist nicht immer eine Pruefsumme: Bei mehrteiligen Uploads
655        setzt S3 einen Hash ueber die Teil-Hashes mit angehaengtem ``-N``.
656        Dann bleibt nur, den Inhalt zu holen und zu vergleichen. Das kostet
657        einen Download - aber nur dann, wenn die Groessen ohnehin gleich sind
658        und die Pruefsumme nichts hergibt.
659        """
660        if oben.size != len(daten):
661            return False
662        bekannt = etag_md5(oben.etag)
663        if bekannt is not None:
664            return bekannt == md5
665        LOG.info("ETag ist keine Pruefsumme (%s), Inhalt wird geholt: %s",
666                 oben.etag, self.s3_path(rel))
667        try:
668            return self.client.get(rel) == daten
669        except Exception as fehler:  # noqa: BLE001 - jede Transportstoerung ist hier gleich
670            raise PublishError(
671                f"Inhalt nicht pruefbar: {self.s3_path(rel)}: {fehler}. Das ETag "
672                "ist keine Pruefsumme, und der Vergleichsdownload scheiterte - "
673                "gleiche Groesse allein gilt nicht als gleicher Inhalt."
674            ) from fehler

731    def _entwerte_entferntes_manifest(self, rel: str) -> None:
732        """Das entfernte Manifest auf ``publishing`` setzen, bevor Daten fallen.
733
734        **Warum das noetig ist.** ``manifest.json`` zuletzt zu schreiben
735        schuetzt einen *neuen* Lauf: Ohne Manifest gilt er nicht. Eine
736        *bestehende* Tabelle schuetzt es nicht. Wird ihre Parquet-Datei
737        ueberschrieben und bricht danach der Manifest-Upload ab, liegt oben
738        der neue Inhalt unter dem alten Manifest - und das sagt weiterhin
739        ``complete``. Genau das war am 15.9.2026 in einer Gegenprobe zu sehen:
740        neue Daten, Bauzeitpunkt vom 1. August, Status ``complete``.
741
742        Deshalb faellt zuerst der Status. Ein Leser sieht danach der Reihe nach
743        den vollstaendigen alten Stand, dann *keinen* (``publishing`` zaehlt
744        nicht als fertig, siehe ``tabellen``), dann den vollstaendigen neuen.
745        Ein Abbruch dazwischen laesst die Tabelle unbrauchbar zurueck, nicht
746        falsch - und der naechste Lauf stellt sie her.
747
748        Was das **nicht** loest: Die fuenf Gold-Tabellen werden weiterhin
749        nacheinander publiziert. Ein gemeinsamer, konsistenter Stand ueber
750        alle Tabellen braucht Versionspfade und einen Verweis, der umgelegt
751        wird; das steht im Ausblick.
752        """
753        if not self.config.publiziert:
754            return
755        schluessel = paths.relativ(rel, MANIFEST)
756        oben = self.client.head(schluessel)
757        if oben is None:
758            return
759        try:
760            manifest = json.loads(self.client.get(schluessel).decode("utf-8"))
761        except (OSError, ValueError) as fehler:
762            raise PublishError(
763                f"Entferntes Manifest nicht lesbar: {self.s3_path(schluessel)}: {fehler}"
764            ) from fehler
765        if str(manifest.get("status", "")).lower() not in FERTIG:
766            return
767        manifest["status"] = "publishing"
768        manifest["superseded_at_utc"] = datetime.now(timezone.utc).isoformat(
769            timespec="seconds").replace("+00:00", "Z")
770        try:
771            self.client.put(schluessel, json.dumps(manifest, indent=2).encode("utf-8"))
772        except Exception as fehler:  # noqa: BLE001 - jede Transportstoerung ist hier gleich
773            raise PublishError(
774                f"Tabelle nicht entwertet, nichts angefasst: "
775                f"{self.s3_path(schluessel)}: {fehler}"
776            ) from fehler
777        LOG.info("Entferntes Manifest auf 'publishing' gesetzt: %s", schluessel)

790    def hole(self, rel: str | Path) -> Path:
791        """Ein Objekt von S3 in die lokale Wurzel - geprueft und unteilbar.
792
793        Liegt es lokal schon identisch, passiert nichts. Geschrieben wird in
794        eine ``.part``-Datei daneben, verglichen (Groesse, dann MD5 gegen das
795        ETag, wo der Dienst eines liefert), dann umbenannt.
796        """

792

794        eine ``.part``-Datei daneben, verglichen (Groesse, dann MD5 gegen das
