census_pipeline.download_tiger

Load Census TIGER/Line geometries into immutable runs.

Zwei Geometriearten, dasselbe Verfahren: tract liefert die Raumeinheit des Vergleichs, place die Stadtgrenzen. Ohne die zweite laesst sich keine Rate je Einwohner belegen - ein Tract weiss nicht, zu welcher Stadt er gehoert, und ein County ist viel groesser als die Stadt darin. Was sich zwischen den beiden unterscheidet, steht in Auslieferung; alles andere ist geteilt.

Jeder Aufruf schreibt einen Lauf nach data/raw/ und data/bronze/ unter einer gemeinsamen run_id, genau wie die Chicago-Ingestion. Raw haelt das ZIP-Archiv Byte fuer Byte; Bronze ist daraus abgeleitetes Parquet mit der Geometrie als WKB.

Der Unterschied zur Ereignis-Ingestion liegt im Aenderungsverhalten der Quelle. Eine TIGER-Vintage erscheint einmal im Jahr und ist danach eingefroren: Es gibt kein updated_on, also auch kein Wasserzeichen, keinen Lookback und keinen Delta-Lauf. An dessen Stelle tritt eine einfachere Frage - ist diese Vintage fuer diesen Bundesstaat schon geladen? Dann gibt es nichts zu tun, solange nicht --force einen neuen Lauf erzwingt.

Vollstaendigkeit wird auch hier nachgewiesen statt behauptet, nur mit anderen Mitteln: Ein ZIP traegt je Mitgliedsdatei eine CRC32-Pruefsumme. Laesst sich das Archiv oeffnen, stimmen alle CRCs und sind die vier Pflichtdateien eines Shapefiles enthalten, dann ist die Auslieferung vollstaendig angekommen. Ein Fingerabdruck ueber die Feldtypen der Quelle macht Schema-Drift zwischen den Vintages sichtbar - GEOIDFQ etwa kam erst spaeter dazu.

149class Auslieferung(NamedTuple):
150    """Was eine TIGER-Geometrieart von der anderen unterscheidet."""

155    stamm: str

158    def archiv(self, vintage: int, state: str) -> str:

164    def schema(self) -> pa.Schema:
165        fields = [pa.field(n, k) for n, k in self.feld_typen.items()]

268def discard_stale_partials(data_root: Path, now: datetime,
269                           aus: Auslieferung = TRACT) -> list[Path]:
270    """Drop work directories a crashed run left behind."""
271    cutoff = (now - timedelta(hours=STALE_PARTIAL_HOURS)).timestamp()
272    removed = []
273    for layer in LAYERS:
274        pattern = f"{layer}/dataset={aus.dataset}/vintage=*/state=*/.*.part"
275        for path in sorted(data_root.glob(pattern)):
276            if path.is_dir() and path.stat().st_mtime < cutoff:
277                shutil.rmtree(path, ignore_errors=True)
278                log.warning("Discarded stale partial run: %s", path)
279                removed.append(path)
280    return removed

283def completed_runs(data_root: Path, vintage: int, state: str,
284                   aus: Auslieferung = TRACT) -> list[Path]:
285    """Bereits veroeffentlichte Laeufe dieser Vintage.

296        for lauf in ablage.laeufe(prefix, "run_*/manifest.json",
297                                  streng=True, fehler=IngestionError)

301def build_session() -> requests.Session:
302    """Create a session that retries throttling and transient server errors."""
303    retry = Retry(
304        total=RETRY_TOTAL,
305        backoff_factor=RETRY_BACKOFF,

307        allowed_methods=frozenset(["GET", "HEAD"]),

311    session.headers["User-Agent"] = USER_AGENT
312    session.mount("https://", HTTPAdapter(max_retries=retry))
313    return session

316def fetch_archive(session: requests.Session, vintage: int, state: str,
317                  aus: Auslieferung = TRACT) -> Archive:
318    """Laedt ein Archiv und weist nach, dass es unversehrt angekommen ist."""
319    url = tiger_url(vintage, state, aus)
320    try:
321        response = session.get(url, timeout=TIMEOUT_SECONDS)
322        response.raise_for_status()
323        payload = response.content
324    except requests.RequestException as error:
325        raise IngestionError(f"Download failed: {error}") from error
326
327    declared = response.headers.get("Content-Length")
328    content_length = int(declared) if declared and declared.isdigit() else None
329    if content_length is not None and content_length != len(payload):
330        raise IngestionError(
331            f"Unvollstaendige Auslieferung: {len(payload):,} von "
332            f"{content_length:,} Byte"
333        )
334
335    members = verify_archive(payload, vintage, state, aus)
336    log.info(
337        "%s geladen: %s Byte, %s Dateien im Archiv",
338        archive_name(vintage, state, aus),
339        f"{len(payload):,}",
340        len(members),
341    )
342    return Archive(
343        payload=payload,
344        url=url,
345        members=members,
346        content_length=content_length,
347        last_modified=response.headers.get("Last-Modified"),
348    )

351def verify_archive(payload: bytes, vintage: int, state: str,
352                   aus: Auslieferung = TRACT) -> list[str]:
353    """Prueft das Archiv gegen seine eigenen Pruefsummen.
354
355    Ein ZIP fuehrt je Mitgliedsdatei eine CRC32 mit. ``testzip`` liest jede
356    Datei und meldet die erste, deren Pruefsumme nicht stimmt. Zusammen mit den
357    Pflichtdateien des Shapefiles ist das der Vollstaendigkeitsnachweis dieser
358    Quelle - sie liefert keine Zeilenzahl, gegen die sich zaehlen liesse.
359    """
360    try:
361        with zipfile.ZipFile(io.BytesIO(payload)) as archive:
362            broken = archive.testzip()
363            if broken is not None:
364                raise IngestionError(f"Falsche Pruefsumme im Archiv: {broken}")
365            members = sorted(info.filename for info in archive.infolist())
366    except zipfile.BadZipFile as error:
367        raise IngestionError(f"Kein lesbares ZIP-Archiv: {error}") from error
368
369    stem = shapefile_member(vintage, state, aus).removesuffix(".shp")
370    fehlend = [
371        suffix for suffix in REQUIRED_MEMBERS if f"{stem}{suffix}" not in members
372    ]
373    if fehlend:
374        raise IngestionError(f"Shapefile unvollstaendig, es fehlen: {fehlend}")
375    return members

378def source_schema(meta: dict[str, Any]) -> tuple[str, dict[str, str]]:
379    """Fingerabdruck ueber die Feldtypen der Quelle.
380
381    Aendert eine spaetere Vintage den Typ eines Feldes, aendert sich der Hash -
382    auch dann, wenn die Ingestion weiterlaeuft. EXTRA_COLUMN faengt neue Felder
383    auf, eine Typaenderung an einem bestehenden Feld wuerde sie nicht bemerken.
384    """
385    fields = {
386        str(name): str(kind)
387        for name, kind in zip(meta.get("fields", []), meta.get("ogr_types", []))
388    }
389    digest = hashlib.sha256(json_text(fields).encode("utf-8")).hexdigest()
390    return digest, fields

393def json_text(value: Any) -> str:
394    """Serialise a value compactly and deterministically."""
395    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"))

398def read_features(archive_path: Path, vintage: int, state: str,
399                  aus: Auslieferung = TRACT) -> tuple[pa.Table, dict]:
400    """Liest das Shapefile direkt aus dem Archiv, ohne es auszupacken.
401
402    GDAL spricht ZIP-Archive ueber das Praefix ``/vsizip/`` an. Die Geometrie
403    kommt dabei schon als WKB, ganz ohne geopandas oder shapely.
404    """
405    member = shapefile_member(vintage, state, aus)
406    source = f"/vsizip/{archive_path}/{member}"
407    try:
408        meta, table = read_arrow(source)
409    except Exception as error:  # pyogrio wirft je nach Ursache Unterschiedliches
410        raise IngestionError(f"Shapefile nicht lesbar: {error}") from error
411
412    crs = str(meta.get("crs") or "")
413    if crs and crs != EXPECTED_CRS:
414        log.warning("Unerwartete CRS: %s statt %s", crs, EXPECTED_CRS)
415    return table, meta

418def bronze_table(source: pa.Table, aus: Auslieferung = TRACT) -> pa.Table:
419    """Bildet die Quelltabelle auf das gepinnte Schema ab, ohne etwas zu verlieren.
420
421    Eine Spalte, die nicht ins Schema passt - unbekannter Name oder anderer Typ -
422    wandert vollstaendig nach EXTRA_COLUMN. Anders als bei JSON ist der Typ
423    innerhalb einer Shapefile-Spalte einheitlich, deshalb wird spaltenweise
424    entschieden und nicht je Wert.
425    """
426    behalten: dict[str, pa.ChunkedArray] = {}
427    extra: dict[str, list[Any]] = {}
428
429    for name in source.schema.names:
430        if name == SOURCE_GEOMETRY_COLUMN:
431            continue
432        passt = name in aus.feld_typen and source.schema.field(name).type.equals(
433            aus.feld_typen[name]
434        )
435        if passt:
436            behalten[name] = source.column(name)
437        else:
438            extra[name] = source.column(name).to_pylist()
439
440    rows = source.num_rows
441    arrays = []
442    for name, kind in aus.feld_typen.items():
443        if name in behalten:
444            arrays.append(behalten[name].combine_chunks())
445        else:
446            arrays.append(pa.nulls(rows, type=kind))
447
448    if SOURCE_GEOMETRY_COLUMN in source.schema.names:
449        arrays.append(source.column(SOURCE_GEOMETRY_COLUMN).combine_chunks())
450    else:
451        arrays.append(pa.nulls(rows, type=pa.binary()))
452
453    arrays.append(pa.array(extra_column(extra, rows), type=pa.string()))
454    return pa.Table.from_arrays(arrays, schema=aus.schema())

457def extra_column(extra: dict[str, list[Any]], rows: int) -> list[str | None]:
458    """Je Zeile ein JSON-Objekt mit allem, was nicht ins Schema passte."""
459    if not extra:
460        return [None] * rows
461    values: list[str | None] = []
462    for index in range(rows):
463        row = {
464            name: column[index]
465            for name, column in extra.items()
466            if column[index] is not None
467        }
468        values.append(json_text(row) if row else None)
469    return values

472def file_hash(path: Path) -> str:
473    digest = hashlib.sha256()
474    with path.open("rb") as file:
475        for chunk in iter(lambda: file.read(1024 * 1024), b""):
476            digest.update(chunk)
477    return digest.hexdigest()

419def bronze_table(source: pa.Table, aus: Auslieferung = TRACT) -> pa.Table:
420    """Bildet die Quelltabelle auf das gepinnte Schema ab, ohne etwas zu verlieren.
421
422    Eine Spalte, die nicht ins Schema passt - unbekannter Name oder anderer Typ -
423    wandert vollstaendig nach EXTRA_COLUMN. Anders als bei JSON ist der Typ
424    innerhalb einer Shapefile-Spalte einheitlich, deshalb wird spaltenweise
425    entschieden und nicht je Wert.
426    """
427    behalten: dict[str, pa.ChunkedArray] = {}
428    extra: dict[str, list[Any]] = {}
429
430    for name in source.schema.names:
431        if name == SOURCE_GEOMETRY_COLUMN:
432            continue
433        passt = name in aus.feld_typen and source.schema.field(name).type.equals(
434            aus.feld_typen[name]
435        )
436        if passt:
437            behalten[name] = source.column(name)
438        else:
439            extra[name] = source.column(name).to_pylist()
440
441    rows = source.num_rows
442    arrays = []
