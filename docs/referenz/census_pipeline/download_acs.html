census_pipeline.download_acs

Load ACS 5-year estimates per census tract into immutable runs.

Die Tabelle liefert die Nenner, die der Staedtevergleich braucht: Ohne Einwohnerzahl je Tract lassen sich nur absolute Fallzahlen vergleichen, keine Raten. Aufgebaut ist der Lauf wie bei download_tiger - Raw und Bronze unter einer gemeinsamen run_id, Partition nach Vintage und Bundesstaat, Manifest je Schicht.

Die Werte behalten den Typ, den die API sendet: alles ist Text, auch Zahlen. Auch die Fehlwert-Kennung -666666666 bleibt unveraendert stehen - sie zu deuten ist Aufgabe der Staging-Schicht, nicht der Ingestion.

Zwei Eigenheiten dieser Quelle sind der Grund fuer eigene Pruefungen:

Erstens antwortet die API ohne Schluessel mit HTTP 200 und einer HTML-Seite "Missing Key" statt mit einem Fehlerstatus. Wer nur den Status prueft, haelt das fuer einen Erfolg und schreibt einen leeren Lauf. Deshalb wird die Antwort selbst geprueft, nicht ihr Status.

Zweitens nennt die API keine erwartete Zeilenzahl. Nachgewiesen wird die Vollstaendigkeit deshalb ueber die Tracts: keine doppelten GEOIDs, und - sofern ein TIGER-Lauf derselben Vintage vorliegt - ein Abgleich der GEOID-Mengen zwischen Geometrie und Attributen. Beide Quellen stammen vom selben Herausgeber und muessen deckungsgleich sein.

 64KEY_PARAMETER = "key"

 67# Die abgefragten Variablen. Die Spaltennamen bleiben die Codes der Quelle -
 68# uebersetzt wird nicht, die Bedeutung haelt das Manifest fest.
 69VARIABLES = {
 70    "B01003_001E": "Einwohner insgesamt",
 71    "B19013_001E": "Median-Haushaltseinkommen (USD)",
 72    "B17001_002E": "Personen unter der Armutsgrenze",
 73}
 74NAME_FIELD = "NAME"
 75GEOGRAPHY_FIELDS = ("state", "county", "tract")
 76# GEOID ist die einzige abgeleitete Spalte: die drei Geografiefelder
 77# aneinandergehaengt. Sie ist der Schluessel zu den TIGER-Geometrien, und die
 78# Quelle liefert die Teile bereits nullgefuellt ("001", "000100"), sodass das
 79# Zusammensetzen verlustfrei und umkehrbar ist. Das Manifest weist sie als
 80# abgeleitet aus.

 82GEOID_LENGTH = 11
 83# Der ACS kennzeichnet nicht schaetzbare Werte mit negativen Sentinels,
 84# etwa das Median-Einkommen eines Tracts ohne Einwohner. Bronze laesst sie

 89RAW_LAYER = "raw"

108def build_schema() -> pa.Schema:
109    """Pin one Parquet schema for every vintage and every state."""
110    fields = [pa.field(GEOID_COLUMN, pa.string()), pa.field(NAME_FIELD, pa.string())]
111    fields += [pa.field(code, pa.string()) for code in VARIABLES]
112    fields += [pa.field(name, pa.string()) for name in GEOGRAPHY_FIELDS]
113    fields.append(pa.field(EXTRA_COLUMN, pa.string()))
114    return pa.schema(fields)

153def census_key() -> str:
154    """Holt den Schluessel aus .env; ohne ihn ist die Quelle nicht nutzbar."""
155    dotenv_path = find_dotenv(usecwd=True)
156    if dotenv_path:

161            f"{KEY_VARIABLE} fehlt. Kostenlos unter "
162            "https://api.census.gov/data/key_signup.html und in .env eintragen."
163        )
164    return key

171def query(state: str) -> list[tuple[str, str]]:
172    """Die Abfrage ohne Schluessel - so steht sie auch im Manifest."""

193def discard_stale_partials(data_root: Path, now: datetime) -> list[Path]:
194    """Drop work directories a crashed run left behind."""
195    cutoff = (now - timedelta(hours=STALE_PARTIAL_HOURS)).timestamp()
196    removed = []
197    for layer in LAYERS:
198        pattern = f"{layer}/dataset={DATASET}/vintage=*/state=*/.*.part"
199        for path in sorted(data_root.glob(pattern)):
200            if path.is_dir() and path.stat().st_mtime < cutoff:
201                shutil.rmtree(path, ignore_errors=True)
202                log.warning("Discarded stale partial run: %s", path)
203                removed.append(path)
204    return removed

218def build_session() -> requests.Session:
219    retry = Retry(
220        total=RETRY_TOTAL,
221        backoff_factor=RETRY_BACKOFF,
222        status_forcelist=RETRY_STATUS,
223        allowed_methods=frozenset(["GET"]),
224        raise_on_status=False,
225    )
226    session = requests.Session()
227    session.headers["User-Agent"] = USER_AGENT
228    session.mount("https://", HTTPAdapter(max_retries=retry))
229    return session

232def fetch_table(
233    session: requests.Session, vintage: int, state: str, key: str
234) -> Answer:
235    """Laedt eine Tabelle und prueft die Antwort, nicht bloss ihren Status."""
236    url = acs_url(vintage)
237    params = [*query(state), (KEY_PARAMETER, key)]
238    try:
239        response = session.get(url, params=params, timeout=TIMEOUT_SECONDS)
240        response.raise_for_status()
241        body = response.content
242    except requests.RequestException as error:
243        raise IngestionError(f"API request failed: {error}") from error
244
245    header, rows = parse_payload(body)
246    log.info("ACS %s %s: %s Tract(s) geladen", vintage, state, f"{len(rows):,}")
247    return Answer(header=header, rows=rows, body=body, url=query_url(vintage, state))

250def parse_payload(body: bytes) -> tuple[list[str], list[list[Any]]]:
251    """Zerlegt die Antwort und faengt die Falle der fehlenden Schluessel ab.
252
253    Die API antwortet ohne Schluessel mit HTTP 200 und einer HTML-Seite. Ein
254    Statuscheck allein wuerde das nicht bemerken, deshalb wird hier auf die
255    erwartete Form bestanden: eine Liste von Listen, deren erste die
256    Spaltennamen enthaelt.
257    """
258    text = body.decode("utf-8", errors="replace").lstrip()
259    if text[:1] != "[":
260        hinweis = "Missing Key" if "Missing Key" in text[:600] else text[:120]
261        raise IngestionError(f"Antwort ist keine ACS-Tabelle: {hinweis}")
262    try:
263        payload = json.loads(text)
264    except json.JSONDecodeError as error:
265        raise IngestionError("API returned invalid JSON") from error
266
267    if not isinstance(payload, list) or len(payload) < 1:
268        raise IngestionError("ACS-Antwort ohne Kopfzeile")
269    header = [str(name) for name in payload[0]]
270    rows = [list(row) for row in payload[1:]]
271    if not rows:
272        raise IngestionError("ACS-Antwort ohne Datenzeilen")
273    fehlend = [
274        field for field in (*GEOGRAPHY_FIELDS, NAME_FIELD) if field not in header
275    ]
276    if fehlend:
277        raise IngestionError(f"ACS-Antwort ohne Pflichtspalten: {fehlend}")
278    return header, rows

281def geoid(row: dict[str, Any]) -> str:
282    """Setzt die Geografiefelder zum Tract-Schluessel zusammen."""
283    value = "".join(str(row.get(field) or "") for field in GEOGRAPHY_FIELDS)
284    if len(value) != GEOID_LENGTH:
285        raise IngestionError(f"Unbrauchbare GEOID: {value!r}")
286    return value

289def bronze_rows(answer: Answer) -> list[dict[str, Any]]:
290    """Bildet die Antwort auf das gepinnte Schema ab, ohne etwas zu verlieren."""
291    bekannt = set(COLUMNS) - {GEOID_COLUMN, EXTRA_COLUMN}
292    zeilen = []
293    for values in answer.rows:
294        record = dict(zip(answer.header, values))
295        row: dict[str, Any] = {GEOID_COLUMN: geoid(record)}
296        for column in COLUMNS:
297            if column in (GEOID_COLUMN, EXTRA_COLUMN):
298                continue
299            value = record.get(column)
300            row[column] = None if value is None else str(value)
301        extra = {
302            name: value
303            for name, value in record.items()
304            if name not in bekannt and value is not None
305        }
306        row[EXTRA_COLUMN] = json_text(extra) if extra else None
307        zeilen.append(row)
308    return zeilen

311def bronze_table(answer: Answer) -> pa.Table:

315def verify_tracts(table: pa.Table) -> list[str]:
316    """Doppelte Tracts waeren ein stiller Fehler; hier brechen sie den Lauf ab."""
317    values = table.column(GEOID_COLUMN).to_pylist()
318    if len(values) != len(set(values)):
319        raise IngestionError("ACS lieferte doppelte GEOIDs")
320    return values

323def tiger_geoids(data_root: Path, vintage: int, state: str) -> list[str] | None:
324    """Die GEOIDs des passenden TIGER-Laufs, falls einer vorliegt.
325
326    Geometrie und Attribute stammen vom selben Herausgeber und derselben
327    Vintage; ihre Tract-Mengen muessen deckungsgleich sein. Fehlt der
328    Geometrie-Lauf, entfaellt der Abgleich - er ist ein Nachweis, kein Zwang.
329    """
330    from census_pipeline import download_tiger
331
332    runs = download_tiger.completed_runs(data_root, vintage, state)
333    if not runs:
334        return None
335    part = runs[-1] / download_tiger.PARQUET_FILE
336    if not part.exists():
337        return None
338    return pq.ParquetFile(part).read().column("GEOID").to_pylist()

341def compare_with_tiger(
342    data_root: Path, vintage: int, state: str, geoids: list[str]
343) -> dict[str, Any]:
344    """Vergleicht die Tract-Mengen und haelt das Ergebnis fest."""
345    referenz = tiger_geoids(data_root, vintage, state)
346    if referenz is None:
347        return {"compared": False, "reason": "kein TIGER-Lauf dieser Vintage"}
348
349    attribute, geometrie = set(geoids), set(referenz)
350    ergebnis = {
351        "compared": True,
352        "tiger_tracts": len(geometrie),
353        "acs_tracts": len(attribute),
354        "matching": attribute == geometrie,
355        "only_in_tiger": sorted(geometrie - attribute)[:20],
356        "only_in_acs": sorted(attribute - geometrie)[:20],
357    }
358    if not ergebnis["matching"]:
359        log.warning(
360            "Tract-Mengen weichen ab: %s nur in TIGER, %s nur in ACS",
361            len(geometrie - attribute),
362            len(attribute - geometrie),
363        )
364    return ergebnis

371def file_hash(path: Path) -> str:
372    digest = hashlib.sha256()
373    with path.open("rb") as file:
374        for chunk in iter(lambda: file.read(1024 * 1024), b""):
375            digest.update(chunk)
376    return digest.hexdigest()

387def write_landing(work_dir: Path, body: bytes) -> list[dict[str, Any]]:

403def write_run(
404    data_root: Path,
405    started: datetime,
406    answer: Answer,
407    *,
408    vintage: int,
409    state: str,
410) -> RunPaths:
411    """Schreibt beide Schichten und veroeffentlicht sie atomar, Raw zuerst."""
412    ingest_date = started.date().isoformat()
413    run_id = started.strftime("run_%Y%m%dT%H%M%S%fZ")
414    parents = {layer: run_parent(data_root, layer, vintage, state) for layer in LAYERS}
415    work = {layer: parent / f".{run_id}.part" for layer, parent in parents.items()}
416    final = {layer: parent / run_id for layer, parent in parents.items()}
417    for path in work.values():
418        path.mkdir(parents=True)
419
420    try:
421        files = {RAW_LAYER: write_landing(work[RAW_LAYER], answer.body)}
422        table, files[BRONZE_LAYER] = write_bronze(work[BRONZE_LAYER], answer)
423        geoids = verify_tracts(table)
424
425        manifest = {
426            "status": "complete",
427            "contract_version": CONTRACT_VERSION,
428            "code_version": code_stand(),
429            "immutable": True,
430            "source": "census_acs5",
431            "dataset": DATASET,
432            "vintage": vintage,
433            "state_fips": state,
434            "state_name": STATES.get(state),
435            "ingest_date": ingest_date,
436            "run_id": run_id,
437            "started_at_utc": started.isoformat().replace("+00:00", "Z"),
438            "completed_at_utc": datetime.now(timezone.utc)
439            .isoformat()
440            .replace("+00:00", "Z"),
441            "row_count": table.num_rows,
442            "variables": dict(VARIABLES),
443            "derived_columns": {
444                GEOID_COLUMN: "state + county + tract, Schluessel zu TIGER"
445            },
446            "missing_value_code": MISSING_VALUE_CODE,
447            "completeness": {
448                # Die Quelle nennt keine Zeilenzahl. Nachgewiesen wird die
449                # Eindeutigkeit der Tracts und, wenn moeglich, die Deckung mit
450                # den Geometrien derselben Vintage.
451                "verified": True,
452                "unique_tracts": len(set(geoids)),
453                "tiger_cross_check": compare_with_tiger(
454                    data_root, vintage, state, geoids
455                ),
456            },
457            "query": {
458                "method": "GET",
459                # Der Schluessel gehoert bei dieser API in die Abfrage, nicht in
460                # einen Header. Im Manifest steht die URL deshalb ohne ihn.
461                "url_without_key": answer.url,
462                "key_parameter": KEY_PARAMETER,
463            },
464        }
465        for layer in LAYERS:
466            document = dict(manifest, layer=layer, format=LAYER_FORMAT[layer])
467            if layer == BRONZE_LAYER:
468                document["schema"] = {
469                    field.name: str(field.type) for field in SCHEMA
470                }
471                document["derived_from"] = f"{RAW_LAYER}/.../{run_id}"
472            document["files"] = files[layer]
473            (work[layer] / "manifest.json").write_text(
474                json.dumps(document, ensure_ascii=False, indent=2) + "\n",
475                encoding="utf-8",
476            )
477
478        work[RAW_LAYER].rename(final[RAW_LAYER])
479        work[BRONZE_LAYER].rename(final[BRONZE_LAYER])
480        return RunPaths(final[RAW_LAYER], final[BRONZE_LAYER])
481    except BaseException:
482        for path in work.values():
483            shutil.rmtree(path, ignore_errors=True)
484        raise

487def ingest_state(
488    session: requests.Session,
489    data_root: Path,
490    started: datetime,
491    vintage: int,
492    state: str,
493    key: str,
494    force: bool,
495) -> RunPaths | None:
496    """Laedt eine Vintage fuer einen Bundesstaat; None, wenn nichts zu tun war."""
497    vorhanden = completed_runs(data_root, vintage, state)
498    if vorhanden and not force:
499        log.info(
500            "ACS %s %s ist bereits geladen (%s); --force erzwingt einen neuen Lauf",
501            vintage,
502            STATES.get(state, state),
503            vorhanden[-1].name,
504        )
505        return None
506    answer = fetch_table(session, vintage, state, key)
507    return write_run(data_root, started, answer, vintage=vintage, state=state)

510def main() -> int:
511    args = arguments()
512    started = datetime.now(timezone.utc)
513
514    ablage = for_root(args.data_root)
515    data_root = ablage.local_root
516
517    try:
518        key = census_key()

def parse_payload(body: bytes) -> tuple[list[str], list[list[typing.Any]]]: 
251def parse_payload(body: bytes) -> tuple[list[str], list[list[Any]]]:
252    """Zerlegt die Antwort und faengt die Falle der fehlenden Schluessel ab.
253
254    Die API antwortet ohne Schluessel mit HTTP 200 und einer HTML-Seite. Ein
255    Statuscheck allein wuerde das nicht bemerken, deshalb wird hier auf die
256    erwartete Form bestanden: eine Liste von Listen, deren erste die
257    Spaltennamen enthaelt.
258    """
259    text = body.decode("utf-8", errors="replace").lstrip()
260    if text[:1] != "[":
261        hinweis = "Missing Key" if "Missing Key" in text[:600] else text[:120]
262        raise IngestionError(f"Antwort ist keine ACS-Tabelle: {hinweis}")
263    try:
264        payload = json.loads(text)
265    except json.JSONDecodeError as error:
266        raise IngestionError("API returned invalid JSON") from error
267
268    if not isinstance(payload, list) or len(payload) < 1:
269        raise IngestionError("ACS-Antwort ohne Kopfzeile")
270    header = [str(name) for name in payload[0]]
271    rows = [list(row) for row in payload[1:]]
272    if not rows:
273        raise IngestionError("ACS-Antwort ohne Datenzeilen")
274    fehlend = [
275        field for field in (*GEOGRAPHY_FIELDS, NAME_FIELD) if field not in header
276    ]
277    if fehlend:
278        raise IngestionError(f"ACS-Antwort ohne Pflichtspalten: {fehlend}")
279    return header, rows

def bronze_rows( answer: Answer) -> list[dict[str, typing.Any]]: 
290def bronze_rows(answer: Answer) -> list[dict[str, Any]]:
291    """Bildet die Antwort auf das gepinnte Schema ab, ohne etwas zu verlieren."""
292    bekannt = set(COLUMNS) - {GEOID_COLUMN, EXTRA_COLUMN}
293    zeilen = []
294    for values in answer.rows:
295        record = dict(zip(answer.header, values))
296        row: dict[str, Any] = {GEOID_COLUMN: geoid(record)}
297        for column in COLUMNS:
298            if column in (GEOID_COLUMN, EXTRA_COLUMN):
299                continue
300            value = record.get(column)
301            row[column] = None if value is None else str(value)
302        extra = {
303            name: value
304            for name, value in record.items()
305            if name not in bekannt and value is not None
306        }
307        row[EXTRA_COLUMN] = json_text(extra) if extra else None
308        zeilen.append(row)
309    return zeilen
